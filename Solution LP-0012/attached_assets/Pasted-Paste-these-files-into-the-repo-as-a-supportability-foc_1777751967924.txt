pub fn fnv1a_32(name: &str) -> [u8; 4] {
    let mut h: u32 = 0x811c9dc5;
    for b in name.as_bytes() {
        h ^= *b as u32;
        h = h.wrapping_mul(0x0100_0193);
    }
    h.to_le_bytes()
}

pub fn encode_event<E: EventSchema>(event: &E) -> EventResult<Vec<u8>> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    if payload.len() > MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: MAX_EVENT_SIZE,
        });
    }

    let mut out = Vec::with_capacity(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(&E::discriminant());
    out.extend_from_slice(&payload);
    Ok(out)
}

pub fn emit<E: EventSchema>(event: &E) -> EventResult<()> {
    let bytes = encode_event(event)?;
    let rc = unsafe { __sys_emit_event(bytes.as_ptr(), bytes.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: bytes.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

#[cfg(feature = "encryption")]
pub fn emit_encrypted<E: EventSchema>(
    event: &E,
    key_bytes: &[u8],
    nonce_bytes: &[u8],
) -> EventResult<()> {
    use aes_gcm::{
        aead::{Aead, KeyInit},
        Aes256Gcm, Key, Nonce,
    };

    if key_bytes.len() != 32 {
        return Err(EventError::InvalidEncryptionKey {
            expected: 32,
            found: key_bytes.len(),
        });
    }
    if nonce_bytes.len() != 12 {
        return Err(EventError::InvalidEncryptionNonce {
            expected: 12,
            found: nonce_bytes.len(),
        });
    }

    let plaintext = encode_event(event)?;
    let key = Key::<Aes256Gcm>::from_slice(key_bytes);
    let nonce = Nonce::from_slice(nonce_bytes);
    let cipher = Aes256Gcm::new(key);

    let ciphertext = cipher
        .encrypt(nonce, plaintext.as_ref())
        .map_err(|_| EventError::EncryptionFailed)?;

    let rc = unsafe { __sys_emit_event(ciphertext.as_ptr(), ciphertext.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: ciphertext.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

use alloc::string::{String, ToString};

pub type EventResult<T> = core::result::Result<T, EventError>;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventError {
    SerializationFailed,
    EventTooLarge { size: usize, limit: usize },
    TxBudgetExceeded { used: usize, added: usize, limit: usize },
    TxCountExceeded { used: usize, limit: usize },
    InvalidEnvelope(String),
    InvalidVersion(u8),
    InvalidEncryptionKey { expected: usize, found: usize },
    InvalidEncryptionNonce { expected: usize, found: usize },
    InvalidHex(String),
    InvalidProgramId(String),
    InvalidTxHash(String),
    MissingField(&'static str),
    UnsupportedFormat(String),
    EncryptionFailed,
    Io(String),
    Rpc(String),
    SyscallError(i32),
}

impl core::fmt::Display for EventError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::SerializationFailed => f.write_str("failed to serialize event"),
            Self::EventTooLarge { size, limit } => {
                write!(f, "event too large: {size} bytes exceeds {limit} bytes")
            }
            Self::TxBudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::TxCountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
            Self::InvalidEnvelope(msg) => write!(f, "invalid envelope: {msg}"),
            Self::InvalidVersion(v) => write!(f, "unsupported event version: {v}"),
            Self::InvalidEncryptionKey { expected, found } => {
                write!(f, "invalid encryption key length: expected {expected}, found {found}")
            }
            Self::InvalidEncryptionNonce { expected, found } => {
                write!(f, "invalid encryption nonce length: expected {expected}, found {found}")
            }
            Self::InvalidHex(s) => write!(f, "invalid hex string: {s}"),
            Self::InvalidProgramId(s) => write!(f, "invalid program id: {s}"),
            Self::InvalidTxHash(s) => write!(f, "invalid transaction hash: {s}"),
            Self::MissingField(name) => write!(f, "missing required field: {name}"),
            Self::UnsupportedFormat(fmt) => write!(f, "unsupported output format: {fmt}"),
            Self::EncryptionFailed => f.write_str("event encryption failed"),
            Self::Io(s) => write!(f, "i/o error: {s}"),
            Self::Rpc(s) => write!(f, "rpc error: {s}"),
            Self::SyscallError(code) => write!(f, "syscall returned error code {code}"),
        }
    }
}

#[derive(Debug, Clone)]
pub struct SupportBundleWriter {
    output_dir: PathBuf,
}

impl SupportBundleWriter {
    pub fn new(output_dir: impl Into<PathBuf>) -> Self {
        Self {
            output_dir: output_dir.into(),
        }
    }

    fn timestamp() -> String {
        let now = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default();
        format!("{}", now.as_secs())
    }

    pub fn write_bundle(&self, bundle: &Bundle) -> Result<PathBuf, EventError> {
        fs::create_dir_all(&self.output_dir).map_err(|e| EventError::Io(e.to_string()))?;
        let file = self.output_dir.join(format!(
            "{}-{:?}-support.json",
            Self::timestamp(),
            bundle.kind
        ));
        let raw = serde_json::to_string_pretty(bundle).map_err(|e| EventError::Io(e.to_string()))?;
        fs::write(&file, raw).map_err(|e| EventError::Io(e.to_string()))?;
        Ok(file)
    }
}

pub fn render_support_report(bundle: &Bundle) -> String {
    let mut out = String::new();
    out.push_str(&format!("Support bundle kind: {:?}\n", bundle.kind));
    out.push_str(&format!("Command: {}\n", bundle.context.command));

    if let Some(ref url) = bundle.context.rpc_url {
        out.push_str(&format!("RPC: {}\n", url));
    }
    if let Some(ref tx) = bundle.context.tx_hash {
        out.push_str(&format!("Tx: {}\n", tx));
    }
    if let Some(ref pid) = bundle.context.program_id {
        out.push_str(&format!("Program: {}\n", pid));
    }
    if let Some(ref rp) = bundle.context.receipt_path {
        out.push_str(&format!("Receipt: {}\n", rp));
    }

    if let Some(ref diag) = bundle.diagnostics {
        out.push_str(&format!("\nDiagnostics: {}\n", diag.title));
        for r in &diag.records {
            out.push_str(&format!(
                "- [{}] {}: {}\n",
                r.level.as_str(),
                r.component,
                r.message
            ));
            if let Some(ref d) = r.detail {
                out.push_str(&format!("  detail: {}\n", d));
            }
            if let Some(ref c) = r.code {
                out.push_str(&format!("  code: {}\n", c));
            }
        }
    }

    if !bundle.health.is_empty() {
        out.push_str("\nHealth:\n");
        for h in &bundle.health {
            out.push_str(&format!(
                "- {} => {}: {}\n",
                h.name,
                h.status.as_str(),
                h.message
            ));
            if let Some(ref remediation) = h.remediation {
                out.push_str(&format!("  remediation: {}\n", remediation));
            }
        }
    }

    out
}

pub fn build_support_bundle(
    cfg: &SupportBundleConfig,
    context: SupportContext,
    receipt: Option<&ReceiptEnvelope>,
) -> Bundle {
    let mut bundle = Bundle::new(BundleKind::Support, context);

    if cfg.include_diagnostics {
        let mut diag = DiagnosticReport::new("Support diagnostics", "support report generated");
        diag.push(
            DiagnosticRecord::new(DiagnosticLevel::Info, "support", "bundle created")
                .with_detail("baseline support bundle assembled"),
        );
        bundle.set_diagnostics(diag);
    }

    if cfg.include_health {
        bundle.push_health(HealthCheck::healthy(
            "config",
            "configuration loaded and parsed successfully",
        ));

        if let Some(receipt) = receipt {
            if receipt.validate().is_ok() {
                bundle.push_health(HealthCheck::healthy(
                    "receipt",
                    "receipt structure passed validation",
                ));
            } else {
                bundle.push_health(HealthCheck::unhealthy(
                    "receipt",
                    "receipt structure failed validation",
                    "re-download or regenerate the receipt file",
                ));
            }
        } else {
            bundle.push_health(HealthCheck::degraded(
                "receipt",
                "no receipt was supplied",
                "provide --file receipt.json or --tx <hash>",
            ));
        }
    }

    if cfg.include_receipt {
        if let Some(receipt) = receipt {
            bundle.push_entry("receipt", "json", serde_json::to_string_pretty(receipt).unwrap_or_default());
        }
    }

    if cfg.include_env {
        let mut count = 0usize;
        for (k, v) in env::vars() {
            if count >= cfg.max_env_vars {
                break;
            }
            bundle.push_entry(
                format!("env-{k}"),
                "env",
                format!("{k}={v}"),
            );
            count += 1;
        }
    }

    bundle
}

}

use anyhow::{Context, Result};
use lez_events::{AppConfig, CliConfig, OutputFormat};
use std::{env, fs, path::PathBuf};

use crate::commands::CommandLine;

pub fn resolve_config(cli: &CommandLine) -> Result<CliConfig> {
    let mut cfg = if let Some(path) = cli.config.as_ref() {
        load_config_file(path)?
    } else if let Ok(path) = env::var("LEZ_EVENTS_CONFIG") {
        load_config_file(PathBuf::from(path))?
    } else {
        CliConfig::default()
    };

    if let Some(rpc) = cli.rpc_url.as_ref() {
        cfg.rpc_url = rpc.clone();
    }

    if cli.json {
        cfg.output = OutputFormat::Json;
    }
    if cli.jsonl {
        cfg.output = OutputFormat::JsonLines;
    }
    if cli.pretty {
        cfg.output = OutputFormat::Pretty;
    }

    if cli.no_color {
        cfg.color = false;
    }
    if cli.strict {
        cfg.strict = true;
    }
    if cli.follow {
        cfg.follow = true;
    }

    cfg.validate().context("invalid configuration")?;
    Ok(cfg)
}

pub fn render_error(err: &dyn std::error::Error) -> String {
    let mut out = err.to_string();
    let mut source = err.source();
    while let Some(next) = source {
        out.push_str("\n  caused by: ");
        out.push_str(&next.to_string());
        source = next.source();
    }
    out
}

fn inspect_command(args: InspectArgs, cfg: CliConfig) -> Result<()> {
    let raw = fs::read_to_string(&args.file)
        .with_context(|| format!("failed to read {}", args.file.display()))?;
    let envelope: ReceiptEnvelope = serde_json::from_str(&raw)
        .with_context(|| format!("failed to parse {}", args.file.display()))?;

    envelope.validate().context("receipt failed validation")?;

    let summary = serde_json::json!({
        "tx_hash": envelope.tx_hash,
        "status": envelope.status,
        "error": envelope.error,
        "state_root": envelope.state_root,
        "event_count": envelope.events.len(),
        "total_event_chars": envelope.total_hex_chars(),

    match cfg.output {
        lez_events::OutputFormat::Pretty => println!("{}", serde_json::to_string_pretty(&summary)?),
        lez_events::OutputFormat::Json | lez_events::OutputFormat::JsonLines => {
            println!("{}", serde_json::to_string(&summary)?)
        }
    }

    Ok(())
}

fn index_command(args: IndexArgs, _cfg: CliConfig) -> Result<()> {
    let input = if let Some(path) = args.file {
        fs::read_to_string(&path).with_context(|| format!("failed to read {}", path.display()))?
    } else {
        prompts::read_stdin_all()?
    };

    let receipts: Vec<ReceiptEnvelope> = if input.trim().starts_with('[') {
        serde_json::from_str(&input)?
    } else {
        vec![serde_json::from_str(&input)?]
    };

    let mut decoded: Vec<DecodedReceipt> = Vec::with_capacity(receipts.len());
    let mut index = EventIndex::new();

    for receipt in receipts.into_iter() {
        receipt.validate().context("receipt failed validation")?;

        let parsed_events = receipt
            .events
            .iter()
            .map(|hex| decode_hex_envelope(hex, None))
            .collect::<Result<Vec<_>, _>>()
            .map_err(anyhow::Error::from)?;

        index.push_receipt(&receipt, &parsed_events);

        decoded.push(DecodedReceipt {
            tx_hash: receipt.tx_hash,
            status: receipt.status,
            error: receipt.error,
            state_root: receipt.state_root,
            events: parsed_events,
        });
    }

    atomic_write_json(&args.out, &decoded)?;
    println!("indexed {} receipts into {}", decoded.len(), args.out.display());
    Ok(())
}

fn query_command(args: QueryArgs, _cfg: CliConfig) -> Result<()> {
    let raw = fs::read_to_string(&args.file)
        .with_context(|| format!("failed to read {}", args.file.display()))?;
    let rows: Vec<DecodedReceipt> = serde_json::from_str(&raw)?;

    let filtered: Vec<_> = rows
        .into_iter()
        .filter(|r| args.tx_hash.as_ref().map_or(true, |wanted| &r.tx_hash == wanted))
        .filter(|r| args.status.as_ref().map_or(true, |wanted| {
            format!("{:?}", r.status).to_lowercase() == wanted.to_lowercase()
        }))
        .filter(|r| args.type_name.as_ref().map_or(true, |wanted| {
            r.events.iter().any(|e| e.type_name.as_deref() == Some(wanted))
        }))
        .collect();

    println!("{}", serde_json::to_string_pretty(&filtered)?);
    Ok(())
}

fn load_receipt_envelope(args: &DecodeArgs, cfg: &CliConfig) -> Result<ReceiptEnvelope> {
    if let Some(path) = &args.file {
        let raw = fs::read_to_string(path)
            .with_context(|| format!("failed to read {}", path.display()))?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)
            .with_context(|| format!("failed to parse {}", path.display()))?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    if args.stdin {
        let raw = prompts::read_stdin_all()?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    if let Some(tx) = &args.tx {
        validate_tx_hash(tx)?;
        let rpc = args.rpc.as_deref().unwrap_or(&cfg.rpc_url);
        let client = Client::builder()
            .timeout(Duration::from_millis(cfg.timeout_ms))
            .build()
            .map_err(|e| EventError::Rpc(e.to_string()))?;
        let url = format!("{}/tx/{}/events", rpc.trim_end_matches('/'), tx);

        let resp = retry(
            RetryConfig {
                attempts: cfg.retries.max(1),
                backoff: lez_events::BackoffConfig::default(),
            },
            || {
                client
                    .get(&url)
                    .send()
                    .map_err(|e| EventError::Rpc(e.to_string()))
            },
        )?;

        if !resp.status().is_success() {
            return Err(EventError::Rpc(format!("rpc returned HTTP {}", resp.status())).into());
        }

        let env: ReceiptEnvelope = resp.json().map_err(|e| EventError::Rpc(e.to_string()))?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    Err(EventError::MissingField("file | stdin | tx").into())
}

#![cfg_attr(not(test), no_std)]

extern crate alloc;

use alloc::string::String;
use alloc::vec;
use alloc::vec::Vec;
use borsh::{BorshDeserialize, BorshSerialize};
use lez_events::{emit_encrypted_event, emit_event, EventError, EventSchema, MAX_EVENT_SIZE};

#[derive(BorshSerialize, BorshDeserialize, Debug)]
pub enum Instruction {
    Success { value: u64 },
    FailAfterEvent { reason: String },
    FailWithoutEvent,
    SizeLimit,
    FriendlyMessage { message: String },
    Private { key: [u8; 32], nonce: [u8; 12], secret: u64 },
}

#[derive(BorshSerialize, BorshDeserialize, Debug, PartialEq, Clone)]
pub struct SuccessEvent {
    pub value: u64,
}

#[derive(BorshSerialize, BorshDeserialize, Debug, Clone)]
pub struct FailureEvent {
    pub reason: String,
}

#[derive(BorshSerialize, BorshDeserialize, Debug, PartialEq, Clone)]
pub struct FriendlyEvent {
    pub message: String,
}

#[derive(BorshSerialize, BorshDeserialize, Debug, PartialEq, Clone)]
pub struct SecretEvent {
    pub secret: u64,
}

#[derive(BorshSerialize, BorshDeserialize, Debug, Clone)]
pub struct OversizedEvent {
    pub data: Vec<u8>,
}

impl EventSchema for SuccessEvent {
    const NAME: &'static str = "event_demo::SuccessEvent";
}
impl EventSchema for FailureEvent {
    const NAME: &'static str = "event_demo::FailureEvent";
}
impl EventSchema for FriendlyEvent {
    const NAME: &'static str = "event_demo::FriendlyEvent";
}
impl EventSchema for SecretEvent {
    const NAME: &'static str = "event_demo::SecretEvent";
}
impl EventSchema for OversizedEvent {
    const NAME: &'static str = "event_demo::OversizedEvent";
}

#[derive(Debug)]
pub enum ProgramError {
    Event(EventError),
    OversizeRequested { requested: usize, limit: usize },
    PanicSimulated(&'static str),
}

impl core::fmt::Display for ProgramError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Event(e) => write!(f, "{e}"),
            Self::OversizeRequested { requested, limit } => {
                write!(f, "requested event size {requested} exceeds limit {limit}")
            }
            Self::PanicSimulated(msg) => write!(f, "simulated failure: {msg}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for ProgramError {}

impl From<EventError> for ProgramError {
    fn from(value: EventError) -> Self {
        Self::Event(value)
    }
}

pub fn process_instruction(instruction: Instruction) -> Result<(), ProgramError> {
    match instruction {
        Instruction::Success { value } => {
            emit_event!(SuccessEvent { value })?;
            Ok(())
        }
        Instruction::FailAfterEvent { reason } => {
            emit_event!(FailureEvent { reason: reason.clone() })?;
            Err(ProgramError::PanicSimulated("intentional failure after event"))
        }
        Instruction::FailWithoutEvent => {
            Err(ProgramError::PanicSimulated("intentional failure before event emission"))
        }
        Instruction::SizeLimit => {
            let payload = vec![0u8; MAX_EVENT_SIZE + 1];
            let oversized = OversizedEvent { data: payload };
            match emit_event!(oversized) {
                Ok(()) => Ok(()),
                Err(EventError::EventTooLarge { size, limit }) => {
                    Err(ProgramError::OversizeRequested { requested: size, limit })
                }
                Err(err) => Err(ProgramError::Event(err)),
            }
        }
        Instruction::FriendlyMessage { message } => {
            emit_event!(FriendlyEvent { message })?;
            Ok(())
        }
        Instruction::Private { key, nonce, secret } => {
            emit_encrypted_event!(SecretEvent { secret }, &key, &nonce)
                .map_err(ProgramError::Event)
        }
    }
}

# docs/supportability.md

## What this adds

This supportability layer is meant to reduce the time between a user report and a reproducible fix.

### New capabilities

- structured diagnostic records
- support bundles containing config, env, and receipts
- health checks for receipt structure and local environment
- an `explain-error` command for user-facing failures
- atomic output writes for safer support artifacts
- clearer CLI context and validation

### Helpful commands

```bash
lez-events doctor --file receipt.json
lez-events health --file receipt.json
lez-events bundle --file receipt.json --out-dir support-bundles
lez-events explain-error --code TxBudgetExceeded --message "event budget exceeded"

Expected support workflow
capture the receipt or failing input
run doctor to validate structure
run health to see pass/fail checks
generate a bundle for sharing
inspect the bundle JSON for environment and diagnostics

This set is aimed at making the code easier to maintain and debug in real use: the CLI now produces better reports, the library can assemble support bundles automatically, and invalid data is rejected with more context.
