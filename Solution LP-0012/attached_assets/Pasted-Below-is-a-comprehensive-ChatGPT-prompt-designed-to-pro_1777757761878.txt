// components/ProofProgress.tsx
"use client";
import { useEffect, useState } from "react";
import { Progress } from "@/components/ui/progress";

interface ProofProgressProps {
  taskId: string;
  estimatedTimeSec: number;
}

export function ProofProgress({ taskId, estimatedTimeSec }: ProofProgressProps) {
  const [progress, setProgress] = useState(0);
  const [elapsed, setElapsed] = useState(0);

  useEffect(() => {
    const interval = setInterval(() => {
      setElapsed(prev => {
        const newElapsed = prev + 0.1;
        const newProgress = Math.min(100, (newElapsed / estimatedTimeSec) * 100);
        setProgress(newProgress);
        return newElapsed;
      });
    }, 100);
    return () => clearInterval(interval);
  }, [estimatedTimeSec]);

  return (
    <div className="space-y-2">
      <div className="flex justify-between text-sm">
        <span>Generating zero‑knowledge proof for ID {taskId}...</span>
        <span>{Math.round(progress)}%</span>
      </div>
      <Progress value={progress} className="w-full" />
    </div>
  );
}
```
