pub fn fnv1a_32(name: &str) -> [u8; 4] {
    let mut h: u32 = 0x811c9dc5;
    for b in name.as_bytes() {
        h ^= *b as u32;
        h = h.wrapping_mul(0x01000193);
    }
    h.to_le_bytes()
}

pub fn encode_event<E: EventSchema>(event: &E) -> EventResult<Vec<u8>> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    if payload.len() > MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: MAX_EVENT_SIZE,
        });
    }

    let mut out = Vec::with_capacity(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(&E::discriminant());
    out.extend_from_slice(&payload);
    Ok(out)
}

pub fn emit<E: EventSchema>(event: &E) -> EventResult<()> {
    let bytes = encode_event(event)?;
    let rc = unsafe { __sys_emit_event(bytes.as_ptr(), bytes.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: bytes.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

#[cfg(feature = "encryption")]
pub fn emit_encrypted<E: EventSchema>(
    event: &E,
    key_bytes: &[u8],
    nonce_bytes: &[u8],
) -> EventResult<()> {
    use aes_gcm::{
        aead::{Aead, KeyInit},
        Aes256Gcm, Key, Nonce,
    };

    if key_bytes.len() != 32 {
        return Err(EventError::InvalidEncryptionKey {
            expected: 32,
            found: key_bytes.len(),
        });
    }
    if nonce_bytes.len() != 12 {
        return Err(EventError::InvalidEncryptionNonce {
            expected: 12,
            found: nonce_bytes.len(),
        });
    }

    let plaintext = encode_event(event)?;
    let key = Key::<Aes256Gcm>::from_slice(key_bytes);
    let nonce = Nonce::from_slice(nonce_bytes);
    let cipher = Aes256Gcm::new(key);

    let ciphertext = cipher
        .encrypt(nonce, plaintext.as_ref())
        .map_err(|_| EventError::EncryptionFailed)?;

    let rc = unsafe { __sys_emit_event(ciphertext.as_ptr(), ciphertext.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: ciphertext.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

pub type EventResult<T> = core::result::Result<T, EventError>;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventError {
    SerializationFailed,
    EventTooLarge { size: usize, limit: usize },
    TxBudgetExceeded { used: usize, added: usize, limit: usize },
    TxCountExceeded { used: usize, limit: usize },
    InvalidEnvelope(String),
    InvalidVersion(u8),
    InvalidEncryptionKey { expected: usize, found: usize },
    InvalidEncryptionNonce { expected: usize, found: usize },
    InvalidHex(String),
    InvalidProgramId(String),
    InvalidTxHash(String),
    MissingField(&'static str),
    UnsupportedFormat(String),
    EncryptionFailed,
    Io(String),
    Rpc(String),
    SyscallError(i32),
}

impl core::fmt::Display for EventError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::SerializationFailed => f.write_str("failed to serialize event"),
            Self::EventTooLarge { size, limit } => {
                write!(f, "event too large: {size} bytes exceeds {limit} bytes")
            }
            Self::TxBudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::TxCountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
            Self::InvalidEnvelope(msg) => write!(f, "invalid envelope: {msg}"),
            Self::InvalidVersion(v) => write!(f, "unsupported event version: {v}"),
            Self::InvalidEncryptionKey { expected, found } => {
                write!(f, "invalid encryption key length: expected {expected}, found {found}")
            }
            Self::InvalidEncryptionNonce { expected, found } => {
                write!(f, "invalid encryption nonce length: expected {expected}, found {found}")
            }
            Self::InvalidHex(s) => write!(f, "invalid hex string: {s}"),
            Self::InvalidProgramId(s) => write!(f, "invalid program id: {s}"),
            Self::InvalidTxHash(s) => write!(f, "invalid transaction hash: {s}"),
            Self::MissingField(name) => write!(f, "missing required field: {name}"),
            Self::UnsupportedFormat(fmt) => write!(f, "unsupported output format: {fmt}"),
            Self::EncryptionFailed => f.write_str("event encryption failed"),
            Self::Io(s) => write!(f, "i/o error: {s}"),
            Self::Rpc(s) => write!(f, "rpc error: {s}"),
            Self::SyscallError(code) => write!(f, "syscall returned error code {code}"),
        }
    }
}

            Self::TooShort { min, found } => write!(f, "too short: minimum {min}, found {found}"),
            Self::TooLong { max, found } => write!(f, "too long: maximum {max}, found {found}"),
            Self::NotHex => f.write_str("value is not valid hex"),

            Self::NotAscii => f.write_str("value contains non-ascii characters"),
        }

pub fn validate_hex_string(input: &str) -> Result<(), EventError> {
    if input.trim().is_empty() {
        return Err(EventError::InvalidHex("empty string".into()));
    }

    let s = input.trim().trim_start_matches("0x");
    if s.len() % 2 != 0 {
        return Err(EventError::InvalidHex(
            "hex string must have an even number of digits".into(),
        ));
    }

    if hex::decode(s).is_err() {
        return Err(EventError::InvalidHex("could not decode hex".into()));
    }

    Ok(())
}

pub fn validate_program_id(input: &str) -> Result<(), EventError> {
    let s = input.trim();
    if s.is_empty() {
        return Err(EventError::MissingField("program_id"));
    }
    let s = s.trim_start_matches("0x");
    if s.len() != 64 {
        return Err(EventError::InvalidProgramId(
            "program id must be 32 bytes hex-encoded".into(),
        ));
    }
    validate_hex_string(input)?;
    Ok(())
}

pub fn validate_tx_hash(input: &str) -> Result<(), EventError> {
    let s = input.trim();
    if s.is_empty() {
        return Err(EventError::MissingField("tx_hash"));
    }
    validate_hex_string(input)?;
    Ok(())
}

pub fn validate_event_bytes(bytes: &[u8]) -> Result<(), EventError> {
    if bytes.is_empty() {
        return Err(EventError::InvalidEnvelope("empty envelope".into()));
    }
    if bytes.len() > 1 + 4 + MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: bytes.len(),
            limit: 1 + 4 + MAX_EVENT_SIZE,
        });
    }
    Ok(())
}

use alloc::string::{String, ToString};

pub fn human_bytes(n: usize) -> String {
    const KB: f64 = 1024.0;
    const MB: f64 = 1024.0 * 1024.0;
    let n_f = n as f64;

    if n_f >= MB {
        format!("{:.2} MiB", n_f / MB)
    } else if n_f >= KB {
        format!("{:.2} KiB", n_f / KB)
    } else {
        format!("{n} B")
    }
}

pub fn human_hex_preview(input: &str, keep: usize) -> String {
    if input.len() <= keep {
        return input.to_string();
    }
    let prefix = &input[..keep.min(input.len())];
    format!("{prefix}… ({} chars total)", input.len())
}

pub fn human_error_chain(err: &dyn core::fmt::Display) -> String {
    err.to_string()
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct DecodedEnvelope {
    pub version: u8,
    pub discriminant: String,
    pub type_name: Option<String>,
    pub payload_hex: String,
    pub payload_size: usize,
    pub raw_size: usize,
}

pub fn decode_envelope(raw: &[u8]) -> Result<DecodedEnvelope, EventError> {
    validate_event_bytes(raw)?;
    let env = crate::decode_envelope(raw)?;
    Ok(DecodedEnvelope {
        version: env.version,
        discriminant: hex::encode(env.discriminant),
        type_name: None,
        payload_hex: hex::encode(&env.payload),
        payload_size: env.payload.len(),
        raw_size: raw.len(),
    })
}

pub fn decode_hex_envelope(
    raw_hex: &str,
    idl: Option<&HashMap<[u8; 4], String>>,
) -> Result<DecodedEnvelope, EventError> {
    let clean = raw_hex.trim().trim_start_matches("0x");
    let raw = hex::decode(clean).map_err(|_| EventError::InvalidHex(raw_hex.to_string()))?;
    let env = crate::decode_envelope(&raw)?;

    let type_name = idl.and_then(|m| m.get(&env.discriminant)).cloned();
    Ok(DecodedEnvelope {
        version: env.version,
        discriminant: hex::encode(env.discriminant),
        type_name,
        payload_hex: hex::encode(&env.payload),
        payload_size: env.payload.len(),
        raw_size: raw.len(),
    })
}

pub fn build_idl(entries: &[(&str, [u8; 4])]) -> HashMap<[u8; 4], String> {
    entries
        .iter()
        .map(|(name, disc)| (*disc, (*name).to_string()))
        .collect()
}

pub fn register_schema<E>(idl: &mut HashMap<[u8; 4], String>)
where
    E: crate::EventSchema,
{
    idl.insert(E::discriminant(), E::NAME.to_string());
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct EventReceipt {
    pub tx_hash: String,
    pub status: ReceiptStatus,
    pub error: Option<String>,
    pub state_root: Option<String>,
    pub events: Vec<String>,

#[derive(Debug, Clone)]
pub struct EventJournal {
    tx_hash: String,
    events: Vec<String>,
    total_bytes: usize,
    count: usize,
    max_bytes: usize,
    max_count: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JournalError {
    BudgetExceeded { used: usize, added: usize, limit: usize },
    CountExceeded { used: usize, limit: usize },
}

impl core::fmt::Display for JournalError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::BudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::CountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
        }
    }
}

impl std::error::Error for JournalError {}

impl EventJournal {
    pub fn new(tx_hash: impl Into<String>) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes: crate::MAX_TX_EVENT_BYTES,
            max_count: crate::MAX_EVENTS_PER_TX,
        }
    }

    pub fn with_limits(tx_hash: impl Into<String>, max_bytes: usize, max_count: usize) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes,
            max_count,
        }
    }

    pub fn push_hex(&mut self, hex_event: String, raw_len: usize) -> Result<(), JournalError> {
        if self.count >= self.max_count {
            return Err(JournalError::CountExceeded {
                used: self.count,
                limit: self.max_count,
            });
        }

        let next = self.total_bytes.saturating_add(raw_len);
        if next > self.max_bytes {
            return Err(JournalError::BudgetExceeded {
                used: self.total_bytes,
                added: raw_len,
                limit: self.max_bytes,
            });
        }

        self.events.push(hex_event);
        self.count += 1;
        self.total_bytes = next;
        Ok(())
    }

    pub fn finish_success(self, state_root: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: ReceiptStatus::Success,
            error: None,
            state_root: Some(state_root.into()),
            events: self.events,
        }
    }

    pub fn finish_failure(self, error: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: ReceiptStatus::Failed,
            error: Some(error.into()),
            state_root: None,
            events: self.events,
        }
    }
}

pub trait ReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError>;
    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError>;
    fn list(&self) -> Result<Vec<EventReceipt>, StoreError>;
}

#[derive(Debug, Clone)]
pub struct JsonFileReceiptStore {
    path: PathBuf,
}

impl JsonFileReceiptStore {
    pub fn new(path: impl Into<PathBuf>) -> Self {
        Self { path: path.into() }
    }

    fn load_all(&self) -> Result<Vec<EventReceipt>, StoreError> {
        if !self.path.exists() {
            return Ok(Vec::new());
        }

        let raw = fs::read_to_string(&self.path).map_err(StoreError::Io)?;
        if raw.trim().is_empty() {
            return Ok(Vec::new());
        }

        serde_json::from_str(&raw).map_err(StoreError::Serde)
    }

    fn save_all(&self, receipts: &[EventReceipt]) -> Result<(), StoreError> {
        if let Some(parent) = self.path.parent() {
            fs::create_dir_all(parent).map_err(StoreError::Io)?;
        }
        let raw = serde_json::to_string_pretty(receipts).map_err(StoreError::Serde)?;
        fs::write(&self.path, raw).map_err(StoreError::Io)
    }
}

impl ReceiptStore for JsonFileReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError> {
        let mut receipts = self.load_all()?;
        if let Some(existing) = receipts.iter_mut().find(|r| r.tx_hash == receipt.tx_hash) {
            *existing = receipt.clone();
        } else {
            receipts.push(receipt.clone());
        }
        self.save_all(&receipts)
    }

    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError> {
        Ok(self.load_all()?.into_iter().find(|r| r.tx_hash == tx_hash))
    }

    fn list(&self) -> Result<Vec<EventReceipt>, StoreError> {
        self.load_all()
    }
}

#[derive(Debug)]
pub enum StoreError {
    Io(std::io::Error),
    Serde(serde_json::Error),
}

impl core::fmt::Display for StoreError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Io(e) => write!(f, "i/o error: {e}"),
            Self::Serde(e) => write!(f, "serialization error: {e}"),
        }
    }
}

impl std::error::Error for StoreError {}

pub fn resolve_config(cli: &CommandLine) -> Result<CliConfig> {

    } else if let Ok(path) = env::var("LEZ_EVENTS_CONFIG") {

        cfg.output = OutputFormat::Pretty;

    }

    if cli.strict {

    }

    if cli.follow {
        cfg.follow = true;
    }

    Ok(cfg)
}

pub fn render_error(err: &dyn std::error::Error) -> String {
    let mut out = err.to_string();
    let mut source = err.source();
    while let Some(next) = source {
        out.push_str("\n  caused by: ");
        out.push_str(&next.to_string());
        source = next.source();
    }
    out
}

fn decode_command(args: DecodeArgs, cfg: CliConfig) -> Result<()> {
    let envelope = load_receipt_envelope(&args, &cfg)?;
    let decoded = decode_receipt(&envelope, &args)?;
    let output = format::render_receipt(&decoded, cfg.output)?;
    println!("{output}");
    Ok(())
}

fn inspect_command(args: InspectArgs, cfg: CliConfig) -> Result<()> {
    let raw = fs::read_to_string(&args.file)
        .with_context(|| format!("failed to read {}", args.file.display()))?;
    let envelope: ReceiptEnvelope = serde_json::from_str(&raw)
        .with_context(|| format!("failed to parse {}", args.file.display()))?;

    let summary = serde_json::json!({
        "tx_hash": envelope.tx_hash,
        "status": envelope.status,
        "error": envelope.error,
        "state_root": envelope.state_root,
        "event_count": envelope.events.len(),
        "total_event_chars": envelope.total_hex_chars(),
    });

    match cfg.output {
        OutputFormat::Pretty => println!("{}", serde_json::to_string_pretty(&summary)?),
        OutputFormat::Json | OutputFormat::JsonLines => println!("{}", serde_json::to_string(&summary)?),
    }

    Ok(())
}

}

        prompts::read_stdin_all()?

fn query_command(args: QueryArgs, _cfg: CliConfig) -> Result<()> {
    let raw = fs::read_to_string(&args.file)
        .with_context(|| format!("failed to read {}", args.file.display()))?;
    let rows: Vec<DecodedReceipt> = serde_json::from_str(&raw)?;

    let filtered: Vec<_> = rows
        .into_iter()
        .filter(|r| args.tx_hash.as_ref().map_or(true, |wanted| &r.tx_hash == wanted))
        .filter(|r| args.status.as_ref().map_or(true, |wanted| format!("{:?}", r.status).to_lowercase() == wanted.to_lowercase()))
        .collect();

    println!("{}", serde_json::to_string_pretty(&filtered)?);
    Ok(())
}

fn load_receipt_envelope(args: &DecodeArgs, cfg: &CliConfig) -> Result<ReceiptEnvelope> {
    if let Some(path) = &args.file {
        let raw = fs::read_to_string(path)
            .with_context(|| format!("failed to read {}", path.display()))?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)
            .with_context(|| format!("failed to parse {}", path.display()))?;
        return Ok(env);
    }

    if args.stdin {
        let raw = prompts::read_stdin_all()?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)?;
        return Ok(env);
    }

    if let Some(tx) = &args.tx {
        let rpc = args.rpc.as_deref().unwrap_or(&cfg.rpc_url);
        let client = Client::new();
        let url = format!("{}/tx/{}/events", rpc.trim_end_matches('/'), tx);
        let resp = client.get(&url).send().map_err(|e| EventError::Rpc(e.to_string()))?;
        if !resp.status().is_success() {
            return Err(EventError::Rpc(format!("rpc returned HTTP {}", resp.status())).into());
        }
        let env: ReceiptEnvelope = resp.json().map_err(|e| EventError::Rpc(e.to_string()))?;
        return Ok(env);
    }

    Err(EventError::MissingField("file | stdin | tx").into())
}

fn decode_receipt(envelope: &ReceiptEnvelope, args: &DecodeArgs) -> Result<DecodedReceipt> {
    let idl: HashMap<[u8; 4], String> = args
        .types
        .iter()
        .map(|t| (lez_events::event_discriminant(t), t.clone()))
        .collect();

    let mut decoded_events = Vec::new();
    for raw_hex in &envelope.events {
        match decode_hex_envelope(raw_hex, Some(&idl)) {
            Ok(e) => decoded_events.push(e),
            Err(err) => {
                if args.raw {
                    decoded_events.push(lez_events::DecodedEnvelope {
                        version: 0,
                        discriminant: "00000000".into(),
                        type_name: Some(format!("decode_error: {}", err)),
                        payload_hex: raw_hex.clone(),
                        payload_size: raw_hex.len() / 2,
                        raw_size: raw_hex.len(),
                    });
                } else {
                    return Err(err.into());
                }
            }
        }
    }

    Ok(DecodedReceipt {
        tx_hash: envelope.tx_hash.clone(),
        status: envelope.status.clone(),
        error: envelope.error.clone(),
        state_root: envelope.state_root.clone(),
        events: decoded_events,
    })
}

pub enum Instruction {

    FailAfterEvent { reason: String },
    FailWithoutEvent,
    SizeLimit,
    FriendlyMessage { message: String },

}

}

}

}

    const NAME: &'static str = "event_demo::SecretEvent";

    const NAME: &'static str = "event_demo::OversizedEvent";

#[derive(Debug)]
pub enum ProgramError {
    Event(EventError),
    OversizeRequested { requested: usize, limit: usize },
    PanicSimulated(&'static str),
}

impl core::fmt::Display for ProgramError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Event(e) => write!(f, "{e}"),
            Self::OversizeRequested { requested, limit } => {
                write!(f, "requested event size {requested} exceeds limit {limit}")
            }
            Self::PanicSimulated(msg) => write!(f, "simulated failure: {msg}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for ProgramError {}

impl From<EventError> for ProgramError {
    fn from(value: EventError) -> Self {
        Self::Event(value)
    }
}

pub fn process_instruction(instruction: Instruction) -> Result<(), ProgramError> {
    match instruction {
        Instruction::Success { value } => {
            emit_event!(SuccessEvent { value })?;
            Ok(())
        }
        Instruction::FailAfterEvent { reason } => {
            emit_event!(FailureEvent { reason: reason.clone() })?;
            Err(ProgramError::PanicSimulated("intentional failure after event"))
        }
        Instruction::FailWithoutEvent => {
            Err(ProgramError::PanicSimulated("intentional failure before event emission"))
        }
        Instruction::SizeLimit => {
            let payload = vec![0u8; MAX_EVENT_SIZE + 1];
            let oversized = OversizedEvent { data: payload };
            match emit_event!(oversized) {
                Ok(()) => Ok(()),
                Err(EventError::EventTooLarge { size, limit }) => {
                    Err(ProgramError::OversizeRequested { requested: size, limit })
                }
                Err(err) => Err(ProgramError::Event(err)),
            }
        }
        Instruction::FriendlyMessage { message } => {
            emit_event!(FriendlyEvent { message })?;
            Ok(())
        }
        Instruction::Private { key, nonce, secret } => {
            emit_encrypted_event!(SecretEvent { secret }, &key, &nonce)
                .map_err(ProgramError::Event)
        }
    }
}

# docs/usage.md

## LEZ Events CLI

The CLI is designed to be usable first:

### Decode a receipt from file
```bash
lez-events decode --file receipt.json

Decode from stdin
cat receipt.json | lez-events decode --stdin

Fetch from RPC
lez-events decode --tx 0xabc123 --rpc http://localhost:8080

Validate user input
lez-events validate --tx-hash 0xabc123 --program-id 0x001122...

Create a config template
lez-events init-config --out lez-events.toml

Inspect receipt summary
lez-events inspect --file receipt.json

Index receipts
lez-events index --file receipts.json --out indexed.json

Query an index
lez-events query --file indexed.json --status failed

Config precedence
Command-line flags
LEZ_EVENTS_CONFIG
Defaults
Output formats
pretty
json
jsonl
Error behavior
The CLI now prefers explicit validation failures over silent assumptions:
malformed hex is rejected with a helpful message
invalid tx hashes are rejected before RPC calls
unsupported envelope versions fail closed
missing inputs are reported clearly

If you want, I can also turn this into a single unified diff against the repo layout so you can paste it into `git apply` directly.
