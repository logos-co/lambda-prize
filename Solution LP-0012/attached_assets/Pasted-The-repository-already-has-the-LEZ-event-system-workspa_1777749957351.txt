#![cfg_attr(not(feature = "std"), no_std)]

extern crate alloc;

use alloc::string::String;
use alloc::vec::Vec;
use borsh::BorshSerialize;

pub const EVENT_VERSION: u8 = 0x00;
pub const MAX_EVENT_SIZE: usize = 64 * 1024;
pub const MAX_TX_EVENT_BYTES: usize = 1024 * 1024;
pub const MAX_EVENTS_PER_TX: usize = 256;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventError {
    SerializationFailed,
    EventTooLarge { size: usize, limit: usize },
    TxBudgetExceeded { used: usize, added: usize, limit: usize },
    TxCountExceeded { used: usize, limit: usize },
    InvalidEnvelope,
    InvalidVersion(u8),
    InvalidProgramId,
    InvalidEncryptionKey,
    InvalidEncryptionNonce,
    EncryptionFailed,
    SyscallError(i32),
}

impl core::fmt::Display for EventError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::SerializationFailed => f.write_str("event serialization failed"),
            Self::EventTooLarge { size, limit } => {
                write!(f, "event too large: {size} bytes exceeds {limit} bytes")
            }
            Self::TxBudgetExceeded { used, added, limit } => {
                write!(f, "transaction event byte budget exceeded: {used} + {added} > {limit}")
            }
            Self::TxCountExceeded { used, limit } => {
                write!(f, "transaction event count exceeded: {used} >= {limit}")
            }
            Self::InvalidEnvelope => f.write_str("invalid event envelope"),
            Self::InvalidVersion(v) => write!(f, "unsupported event version: {v}"),
            Self::InvalidProgramId => f.write_str("invalid program id"),
            Self::InvalidEncryptionKey => f.write_str("invalid encryption key"),
            Self::InvalidEncryptionNonce => f.write_str("invalid encryption nonce"),
            Self::EncryptionFailed => f.write_str("event encryption failed"),
            Self::SyscallError(code) => write!(f, "sys_emit_event failed with code {code}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for EventError {}

pub trait EventSchema: BorshSerialize {
    const NAME: &'static str;

    fn discriminant() -> [u8; 4] {
        fnv1a_32(Self::NAME)
    }
}

pub fn fnv1a_32(name: &str) -> [u8; 4] {
    let bytes = name.as_bytes();
    let mut hash: u32 = 0x811c9dc5;
    let mut i = 0usize;
    while i < bytes.len() {
        hash ^= bytes[i] as u32;
        hash = hash.wrapping_mul(0x01000193);
        i += 1;
    }
    hash.to_le_bytes()
}

pub fn encode_event<E: EventSchema>(event: &E) -> Result<Vec<u8>, EventError> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    if payload.len() > MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: MAX_EVENT_SIZE,
        });
    }

    let mut out = Vec::with_capacity(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(&E::discriminant());
    out.extend_from_slice(&payload);
    Ok(out)
}

pub fn decode_envelope(raw: &[u8]) -> Result<EventEnvelope, EventError> {
    if raw.len() < 5 {
        return Err(EventError::InvalidEnvelope);
    }
    let version = raw[0];
    if version != EVENT_VERSION {
        return Err(EventError::InvalidVersion(version));
    }
    let discriminant: [u8; 4] = raw[1..5]
        .try_into()
        .map_err(|_| EventError::InvalidEnvelope)?;
    let payload = raw[5..].to_vec();
    Ok(EventEnvelope {
        version,
        discriminant,
        payload,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct EventEnvelope {
    pub version: u8,
    pub discriminant: [u8; 4],
    pub payload: Vec<u8>,
}

pub fn emit<E: EventSchema>(event: &E) -> Result<(), EventError> {
    let bytes = encode_event(event)?;
    let rc = unsafe { __sys_emit_event(bytes.as_ptr(), bytes.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: bytes.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

#[cfg(feature = "encryption")]
pub fn emit_encrypted<E: EventSchema>(
    event: &E,
    key_bytes: &[u8],
    nonce_bytes: &[u8],
) -> Result<(), EventError> {
    use aes_gcm::{
        aead::{Aead, KeyInit},
        Aes256Gcm, Key, Nonce,
    };

    if key_bytes.len() != 32 {
        return Err(EventError::InvalidEncryptionKey);
    }
    if nonce_bytes.len() != 12 {
        return Err(EventError::InvalidEncryptionNonce);
    }

    let plaintext = encode_event(event)?;
    let key = Key::<Aes256Gcm>::from_slice(key_bytes);
    let nonce = Nonce::from_slice(nonce_bytes);
    let cipher = Aes256Gcm::new(key);
    let ciphertext = cipher
        .encrypt(nonce, plaintext.as_ref())
        .map_err(|_| EventError::EncryptionFailed)?;

    let rc = unsafe { __sys_emit_event(ciphertext.as_ptr(), ciphertext.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: ciphertext.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

    E: crate::EventSchema,

            event,
            allow_oversize: false,

        self

                size: bytes.len(),
                limit: 1 + 4 + crate::MAX_EVENT_SIZE,
            });
        }
        Ok(bytes)
    }

pub fn decode_raw(bytes: &[u8], idl: Option<&HashMap<[u8; 4], String>>) -> Result<DecodedEvent, EventError> {
    let env = decode_envelope(bytes)?;
    let type_name = idl.and_then(|m| m.get(&env.discriminant)).cloned();

    Ok(DecodedEvent {
        version: env.version,
        discriminant: env.discriminant,
        type_name,
        program_id: None,
        payload_hex: hex::encode(&env.payload),
        payload: env.payload,
    })
}

    entries
        .iter()
        .map(|(name, disc)| (*disc, (*name).to_string()))
        .collect()
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct LogEntry {
    pub tx_hash: String,
    pub status: String,
    pub error: Option<String>,
    pub events: Vec<String>,
}

pub fn append_jsonl(path: &Path, entry: &LogEntry) -> std::io::Result<()> {
    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }

    let mut file = OpenOptions::new().create(true).append(true).open(path)?;
    serde_json::to_writer(&mut file, entry)?;
    file.write_all(b"\n")?;
    file.flush()
}

pub fn read_jsonl(path: &Path) -> std::io::Result<Vec<LogEntry>> {
    let file = File::open(path)?;
    let reader = BufReader::new(file);
    let mut out = Vec::new();

    for line in reader.lines() {
        let line = line?;
        if line.trim().is_empty() {
            continue;
        }
        let parsed: LogEntry = serde_json::from_str(&line)
            .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e))?;
        out.push(parsed);
    }

    Ok(out)
}

// crates/lez-events/src/merkle.rs
use alloc::vec::Vec;
use sha2::{Digest, Sha256};

pub fn sha256(bytes: &[u8]) -> [u8; 32] {
    let mut hasher = Sha256::new();
    hasher.update(bytes);
    hasher.finalize().into()
}

pub fn merkle_root(leaves: &[Vec<u8>]) -> [u8; 32] {
    if leaves.is_empty() {
        return [0u8; 32];
    }

    let mut level: Vec<[u8; 32]> = leaves.iter().map(|leaf| sha256(leaf)).collect();

    while level.len() > 1 {
        if level.len() % 2 == 1 {
            let last = *level.last().unwrap();
            level.push(last);
        }

        let mut next = Vec::with_capacity(level.len() / 2);
        let mut i = 0usize;
        while i < level.len() {
            let mut buf = Vec::with_capacity(64);
            buf.extend_from_slice(&level[i]);
            buf.extend_from_slice(&level[i + 1]);
            next.push(sha256(&buf));
            i += 2;
        }
        level = next;
    }

    level[0]
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub enum TransactionStatus {
    Success,
    Failed,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct EventReceipt {
    pub tx_hash: String,
    pub status: TransactionStatus,
    pub error: Option<String>,
    pub state_root: Option<String>,
    pub events: Vec<String>,
}

#[derive(Debug, Clone)]
pub struct EventJournal {
    tx_hash: String,
    events: Vec<String>,
    total_bytes: usize,
    count: usize,
    max_bytes: usize,
    max_count: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JournalError {
    BudgetExceeded {
        used: usize,
        added: usize,
        limit: usize,
    },
    CountExceeded {
        used: usize,
        limit: usize,
    },
}

impl core::fmt::Display for JournalError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::BudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::CountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
        }
    }
}

impl std::error::Error for JournalError {}

impl EventJournal {
    pub fn new(tx_hash: impl Into<String>) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes: crate::MAX_TX_EVENT_BYTES,
            max_count: crate::MAX_EVENTS_PER_TX,
        }
    }

    pub fn with_limits(tx_hash: impl Into<String>, max_bytes: usize, max_count: usize) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes,
            max_count,
        }
    }

    pub fn push_hex(&mut self, hex_event: String, raw_len: usize) -> Result<(), JournalError> {
        if self.count >= self.max_count {
            return Err(JournalError::CountExceeded {
                used: self.count,
                limit: self.max_count,
            });
        }

        let next_total = self.total_bytes.saturating_add(raw_len);
        if next_total > self.max_bytes {
            return Err(JournalError::BudgetExceeded {
                used: self.total_bytes,
                added: raw_len,
                limit: self.max_bytes,
            });
        }

        self.events.push(hex_event);
        self.count += 1;
        self.total_bytes = next_total;
        Ok(())
    }

    pub fn finish_success(self, state_root: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: TransactionStatus::Success,
            error: None,
            state_root: Some(state_root.into()),
            events: self.events,
        }
    }

    pub fn finish_failure(self, error: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: TransactionStatus::Failed,
            error: Some(error.into()),
            state_root: None,
            events: self.events,
        }
    }
}

pub trait ReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError>;
    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError>;
    fn list(&self) -> Result<Vec<EventReceipt>, StoreError>;
}

#[derive(Debug, Clone)]
pub struct JsonFileReceiptStore {
    path: PathBuf,
}

impl JsonFileReceiptStore {
    pub fn new(path: impl Into<PathBuf>) -> Self {
        Self { path: path.into() }
    }

    fn load_all(&self) -> Result<Vec<EventReceipt>, StoreError> {
        if !self.path.exists() {
            return Ok(Vec::new());
        }

        let raw = fs::read_to_string(&self.path).map_err(StoreError::Io)?;
        if raw.trim().is_empty() {
            return Ok(Vec::new());
        }

        serde_json::from_str(&raw).map_err(StoreError::Serde)
    }

    fn save_all(&self, receipts: &[EventReceipt]) -> Result<(), StoreError> {
        if let Some(parent) = self.path.parent() {
            fs::create_dir_all(parent).map_err(StoreError::Io)?;
        }
        let raw = serde_json::to_string_pretty(receipts).map_err(StoreError::Serde)?;
        fs::write(&self.path, raw).map_err(StoreError::Io)
    }
}

impl ReceiptStore for JsonFileReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError> {
        let mut receipts = self.load_all()?;
        if let Some(existing) = receipts.iter_mut().find(|r| r.tx_hash == receipt.tx_hash) {
            *existing = receipt.clone();
        } else {
            receipts.push(receipt.clone());
        }
        self.save_all(&receipts)
    }

    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError> {
        Ok(self.load_all()?.into_iter().find(|r| r.tx_hash == tx_hash))
    }

    fn list(&self) -> Result<Vec<EventReceipt>, StoreError> {
        self.load_all()
    }
}

#[derive(Debug)]
pub enum StoreError {
    Io(std::io::Error),
    Serde(serde_json::Error),
}

impl core::fmt::Display for StoreError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Io(e) => write!(f, "io error: {e}"),
            Self::Serde(e) => write!(f, "serialization error: {e}"),
        }
    }
}

impl std::error::Error for StoreError {}

        #[arg(long)]
        status: Option<String>,
    },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct RawReceiptFile {
    tx_hash: String,
    status: String,
    error: Option<String>,
    state_root: Option<String>,
    events: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct DecodedReceipt {
    tx_hash: String,
    status: String,
    error: Option<String>,
    state_root: Option<String>,
    events: Vec<serde_json::Value>,
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cli = Cli::parse();

    match cli.command {
        Commands::Decode {
            file,
            tx,
            rpc,
            pretty,
            types,
        } => {
            let receipt = load_receipt(file, tx, rpc)?;
            let idl = build_idl_map(&types);
            let decoded = decode_receipt(&receipt, &idl)?;
            let rendered = if pretty {
                serde_json::to_string_pretty(&decoded)?
            } else {
                serde_json::to_string(&decoded)?
            };
            println!("{rendered}");
        }
        Commands::Index { rpc, out, interval } => {
            run_indexer(&rpc, &out, interval)?;
        }
        Commands::Query { file, tx, status } => {
            let rows = read_index(&file)?;
            let filtered: Vec<_> = rows
                .into_iter()
                .filter(|r: &DecodedReceipt| tx.as_ref().map_or(true, |wanted| &r.tx_hash == wanted))
                .filter(|r| status.as_ref().map_or(true, |wanted| &r.status == wanted))
                .collect();

            println!("{}", serde_json::to_string_pretty(&filtered)?);
        }
    }

    Ok(())
}

fn load_receipt(
    file: Option<PathBuf>,
    tx: Option<String>,
    rpc: Option<String>,
) -> Result<RawReceiptFile, Box<dyn std::error::Error>> {
    if let Some(path) = file {
        let raw = fs::read_to_string(path)?;
        return Ok(serde_json::from_str(&raw)?);
    }

    let tx = tx.ok_or("missing --tx")?;
    let rpc = rpc.ok_or("missing --rpc")?;

    let url = format!("{}/tx/{}/events", rpc.trim_end_matches('/'), tx);
    let resp = reqwest::blocking::get(url)?;
    if !resp.status().is_success() {
        return Err(format!("rpc returned non-success status: {}", resp.status()).into());
    }
    Ok(resp.json()?)
}

fn build_idl_map(types: &[String]) -> HashMap<[u8; 4], String> {
    let mut map = HashMap::new();
    for ty in types {
        map.insert(lez_events::fnv1a_32(ty), ty.clone());
    }
    map
}

fn decode_receipt(
    receipt: &RawReceiptFile,
    idl: &HashMap<[u8; 4], String>,
) -> Result<DecodedReceipt, Box<dyn std::error::Error>> {
    let mut events = Vec::new();

    for raw_hex in &receipt.events {
        let bytes = match hex::decode(raw_hex) {
            Ok(v) => v,
            Err(_) => {
                events.push(serde_json::json!({
                    "error": "invalid hex-encoded event envelope",
                    "raw": raw_hex
                }));
                continue;
            }
        };

        match decode_raw(&bytes, Some(idl)) {
            Ok(decoded) => {
                let value = serde_json::json!({
                    "version": decoded.version,
                    "discriminant": hex::encode(decoded.discriminant),
                    "type_name": decoded.type_name,
                    "program_id": decoded.program_id,
                    "payload_hex": decoded.payload_hex
                });
                events.push(value);
            }
            Err(err) => {
                events.push(serde_json::json!({
                    "error": err.to_string(),
                    "raw": raw_hex
                }));
            }
        }
    }

    Ok(DecodedReceipt {
        tx_hash: receipt.tx_hash.clone(),
        status: receipt.status.clone(),
        error: receipt.error.clone(),
        state_root: receipt.state_root.clone(),
        events,
    })
}

fn run_indexer(
    rpc: &str,
    out: &PathBuf,
    interval: u64,
) -> Result<(), Box<dyn std::error::Error>> {
    let _ = interval;
    if !out.exists() {
        fs::write(out, "[]")?;
    }
    println!("indexer stub started against {rpc}, writing to {out:?}");
    Ok(())
}

fn read_index(path: &PathBuf) -> Result<Vec<DecodedReceipt>, Box<dyn std::error::Error>> {
    if !path.exists() {
        return Ok(Vec::new());
    }
    let raw = fs::read_to_string(path)?;
    if raw.trim().is_empty() {
        return Ok(Vec::new());
    }
    Ok(serde_json::from_str(&raw)?)
}

#[derive(Debug)]
pub enum ProgramError {
    Event(EventError),
    OversizeRequested { requested: usize, limit: usize },
    PanicSimulated(&'static str),
}

impl core::fmt::Display for ProgramError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Event(e) => write!(f, "{e}"),
            Self::OversizeRequested { requested, limit } => {
                write!(f, "requested event size {requested} exceeds limit {limit}")
            }
            Self::PanicSimulated(msg) => write!(f, "simulated failure: {msg}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for ProgramError {}

impl From<EventError> for ProgramError {
    fn from(value: EventError) -> Self {
        Self::Event(value)
    }
}

pub fn process_instruction(instruction: Instruction) -> Result<(), ProgramError> {
    match instruction {
        Instruction::Success { value } => {
            emit_event!(SuccessEvent { value })?;
            Ok(())
        }
        Instruction::FailAfterEvent { reason } => {
            emit_event!(FailureEvent { reason: reason.clone() })?;
            Err(ProgramError::PanicSimulated("intentional failure after event"))
        }
        Instruction::FailWithoutEvent => {
            Err(ProgramError::PanicSimulated("intentional failure before event emission"))
        }
        Instruction::SizeLimit => {
            let payload = vec![0u8; MAX_EVENT_SIZE + 1];
            let oversized = OversizedEvent { data: payload };
            match emit_event!(oversized) {
                Ok(()) => Ok(()),
                Err(EventError::EventTooLarge { size, limit }) => {
                    Err(ProgramError::OversizeRequested { requested: size, limit })
                }
                Err(err) => Err(ProgramError::Event(err)),
            }
        }
        Instruction::Private { key, nonce, secret } => {
            emit_encrypted_event!(SecretEvent { secret }, &key, &nonce).map_err(ProgramError::Event)
        }
    }
}

# LEZ Event Format Specification v1.2

## Overview

A LEZ event is a deterministic, versioned envelope written to transaction receipts and indexed data stores. The runtime must preserve emitted events even if the transaction later fails.

## Envelope

```text
[program_id: 32 bytes]
[version: 1 byte]
[discriminant: 4 bytes]
[payload: N bytes]

Rules
program_id is added by the runtime.
version is 0x00 in this revision.
discriminant is fnv1a_32(type_name).
payload is Borsh-serialized.
The encoder must reject payloads larger than 64 KiB.
The runtime must reject transactions that exceed 1 MiB of total event bytes or 256 events.
Error behavior
Too-large single event: reject the emission immediately.
Transaction byte budget exhausted: reject the emission immediately.
Transaction event count exceeded: reject the emission immediately.
Invalid envelope: return a decode error.
Unknown version: return a decode error and fail closed.
Receipt schema
{
  "tx_hash": "0x...",
  "status": "success | failed",
  "error": "optional string",
  "state_root": "optional string",
  "events": [
    "hex-encoded event envelope"
  ]
}

Decoder guidance
Clients should:
Decode the hex envelope.
Read the version byte.
Read the discriminant.
Use the known schema to parse the payload.
Treat unknown versions as opaque.
Privacy note
Encrypted events may still use the same outer envelope. The payload bytes can be ciphertext instead of plaintext if the program elects to encrypt them.
