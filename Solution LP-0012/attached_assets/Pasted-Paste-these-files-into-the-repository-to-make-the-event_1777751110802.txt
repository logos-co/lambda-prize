pub fn fnv1a_32(name: &str) -> [u8; 4] {
    let mut h: u32 = 0x811c9dc5;
    for b in name.as_bytes() {
        h ^= *b as u32;
        h = h.wrapping_mul(0x0100_0193);
    }
    h.to_le_bytes()
}

pub fn emit<E: EventSchema>(event: &E) -> EventResult<()> {
    let bytes = encode_event(event)?;
    let rc = unsafe { __sys_emit_event(bytes.as_ptr(), bytes.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: bytes.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

#[cfg(feature = "encryption")]
pub fn emit_encrypted<E: EventSchema>(
    event: &E,
    key_bytes: &[u8],
    nonce_bytes: &[u8],
) -> EventResult<()> {
    use aes_gcm::{
        aead::{Aead, KeyInit},
        Aes256Gcm, Key, Nonce,
    };

    if key_bytes.len() != 32 {
        return Err(EventError::InvalidEncryptionKey {
            expected: 32,
            found: key_bytes.len(),
        });
    }
    if nonce_bytes.len() != 12 {
        return Err(EventError::InvalidEncryptionNonce {
            expected: 12,
            found: nonce_bytes.len(),
        });
    }

    let plaintext = encode_event(event)?;
    let key = Key::<Aes256Gcm>::from_slice(key_bytes);
    let nonce = Nonce::from_slice(nonce_bytes);
    let cipher = Aes256Gcm::new(key);

    let ciphertext = cipher
        .encrypt(nonce, plaintext.as_ref())
        .map_err(|_| EventError::EncryptionFailed)?;

    let rc = unsafe { __sys_emit_event(ciphertext.as_ptr(), ciphertext.len()) };
    match rc {
        0 => Ok(()),
        -1 => Err(EventError::TxBudgetExceeded {
            used: 0,
            added: ciphertext.len(),
            limit: MAX_TX_EVENT_BYTES,
        }),
        -2 => Err(EventError::TxCountExceeded {
            used: 0,
            limit: MAX_EVENTS_PER_TX,
        }),
        code => Err(EventError::SyscallError(code)),
    }
}

use alloc::string::{String, ToString};

pub type EventResult<T> = core::result::Result<T, EventError>;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventError {
    SerializationFailed,
    EventTooLarge { size: usize, limit: usize },
    TxBudgetExceeded { used: usize, added: usize, limit: usize },
    TxCountExceeded { used: usize, limit: usize },
    InvalidEnvelope(String),
    InvalidVersion(u8),
    InvalidEncryptionKey { expected: usize, found: usize },
    InvalidEncryptionNonce { expected: usize, found: usize },
    InvalidHex(String),
    InvalidProgramId(String),
    InvalidTxHash(String),
    MissingField(&'static str),
    UnsupportedFormat(String),
    EncryptionFailed,
    Io(String),
    Rpc(String),
    SyscallError(i32),
}

impl core::fmt::Display for EventError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::SerializationFailed => f.write_str("failed to serialize event"),
            Self::EventTooLarge { size, limit } => {
                write!(f, "event too large: {size} bytes exceeds {limit} bytes")
            }
            Self::TxBudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::TxCountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
            Self::InvalidEnvelope(msg) => write!(f, "invalid envelope: {msg}"),
            Self::InvalidVersion(v) => write!(f, "unsupported event version: {v}"),
            Self::InvalidEncryptionKey { expected, found } => {
                write!(f, "invalid encryption key length: expected {expected}, found {found}")
            }
            Self::InvalidEncryptionNonce { expected, found } => {
                write!(f, "invalid encryption nonce length: expected {expected}, found {found}")
            }
            Self::InvalidHex(s) => write!(f, "invalid hex string: {s}"),
            Self::InvalidProgramId(s) => write!(f, "invalid program id: {s}"),
            Self::InvalidTxHash(s) => write!(f, "invalid transaction hash: {s}"),
            Self::MissingField(name) => write!(f, "missing required field: {name}"),
            Self::UnsupportedFormat(fmt) => write!(f, "unsupported output format: {fmt}"),
            Self::EncryptionFailed => f.write_str("event encryption failed"),
            Self::Io(s) => write!(f, "i/o error: {s}"),
            Self::Rpc(s) => write!(f, "rpc error: {s}"),
            Self::SyscallError(code) => write!(f, "syscall returned error code {code}"),
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ValidationIssue {
    Empty,
    TooShort { min: usize, found: usize },
    TooLong { max: usize, found: usize },
    NotHex,
    WrongPrefix,
    NotAscii,
}

impl core::fmt::Display for ValidationIssue {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Empty => f.write_str("value is empty"),
            Self::TooShort { min, found } => write!(f, "too short: minimum {min}, found {found}"),
            Self::TooLong { max, found } => write!(f, "too long: maximum {max}, found {found}"),
            Self::NotHex => f.write_str("value is not valid hex"),
            Self::WrongPrefix => f.write_str("value has the wrong prefix"),
            Self::NotAscii => f.write_str("value contains non-ascii characters"),
        }
    }
}

pub fn validate_hex_string(input: &str) -> Result<(), EventError> {
    let trimmed = input.trim();
    if trimmed.is_empty() {
        return Err(EventError::InvalidHex("empty string".into()));
    }

    let s = trimmed.strip_prefix("0x").unwrap_or(trimmed);
    if s.len() % 2 != 0 {
        return Err(EventError::InvalidHex(
            "hex string must have an even number of digits".into(),
        ));
    }

    if hex::decode(s).is_err() {
        return Err(EventError::InvalidHex("could not decode hex".into()));
    }

    Ok(())
}

pub fn validate_program_id(input: &str) -> Result<(), EventError> {
    let s = input.trim();
    if s.is_empty() {
        return Err(EventError::MissingField("program_id"));
    }
    let s = s.strip_prefix("0x").unwrap_or(s);
    if s.len() != 64 {
        return Err(EventError::InvalidProgramId(
            "program id must be 32 bytes hex-encoded".into(),
        ));
    }
    validate_hex_string(input)?;
    Ok(())
}

pub fn validate_tx_hash(input: &str) -> Result<(), EventError> {
    let s = input.trim();
    if s.is_empty() {
        return Err(EventError::MissingField("tx_hash"));
    }
    validate_hex_string(input)?;
    Ok(())
}

pub fn validate_event_bytes(bytes: &[u8]) -> Result<(), EventError> {
    if bytes.is_empty() {
        return Err(EventError::InvalidEnvelope("empty envelope".into()));
    }
    if bytes.len() > 1 + 4 + MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: bytes.len(),
            limit: 1 + 4 + MAX_EVENT_SIZE,
        });
    }
    Ok(())
}

#![cfg_attr(not(feature = "std"), no_std)]

use alloc::string::String;
use alloc::vec::Vec;
use borsh::BorshSerialize;
use serde::{Deserialize, Serialize};

use crate::{validate_event_bytes, EventError, EventSchema, EVENT_VERSION};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct EnvelopeEncoding {
    pub version: u8,
    pub discriminant: [u8; 4],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DecodedEnvelopeRef<'a> {
    pub version: u8,
    pub discriminant: [u8; 4],
    pub payload: &'a [u8],
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct DecodedEnvelope {
    pub version: u8,
    pub discriminant: String,
    pub type_name: Option<String>,
    pub payload_hex: String,
    pub payload_size: usize,
    pub raw_size: usize,
}

pub fn encode_event<E: EventSchema>(event: &E) -> Result<Vec<u8>, EventError> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    encode_event_bytes(&E::discriminant(), &payload)
}

pub fn encode_event_into<E: EventSchema>(event: &E, out: &mut Vec<u8>) -> Result<(), EventError> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    encode_event_bytes_into(&E::discriminant(), &payload, out)
}

#[inline]
pub fn encode_event_bytes(discriminant: &[u8; 4], payload: &[u8]) -> Result<Vec<u8>, EventError> {
    if payload.len() > crate::MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: crate::MAX_EVENT_SIZE,
        });
    }

    let mut out = Vec::with_capacity(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(discriminant);
    out.extend_from_slice(payload);
    Ok(out)
}

#[inline]
pub fn encode_event_bytes_into(
    discriminant: &[u8; 4],
    payload: &[u8],
    out: &mut Vec<u8>,
) -> Result<(), EventError> {
    if payload.len() > crate::MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: crate::MAX_EVENT_SIZE,
        });
    }

    out.clear();
    out.reserve_exact(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(discriminant);
    out.extend_from_slice(payload);
    Ok(())
}

#[inline]
pub fn decode_envelope(raw: &[u8]) -> Result<DecodedEnvelopeRef<'_>, EventError> {
    validate_event_bytes(raw)?;
    if raw[0] != EVENT_VERSION {
        return Err(EventError::InvalidVersion(raw[0]));
    }

    let discriminant: [u8; 4] = raw[1..5]
        .try_into()
        .map_err(|_| EventError::InvalidEnvelope("missing discriminant".into()))?;

    Ok(DecodedEnvelopeRef {
        version: raw[0],
        discriminant,
        payload: &raw[5..],
    })
}

#[inline]
pub fn decode_hex_envelope(
    raw_hex: &str,
    idl: Option<&std::collections::HashMap<[u8; 4], String>>,
) -> Result<DecodedEnvelope, EventError> {
    let clean = raw_hex.trim().trim_start_matches("0x");
    let raw = hex::decode(clean).map_err(|_| EventError::InvalidHex(raw_hex.to_string()))?;
    let env = decode_envelope(&raw)?;
    Ok(DecodedEnvelope {
        version: env.version,
        discriminant: hex::encode(env.discriminant),
        type_name: idl.and_then(|m| m.get(&env.discriminant)).cloned(),
        payload_hex: hex::encode(env.payload),
        payload_size: env.payload.len(),
        raw_size: raw.len(),
    })
}

#[inline]
pub fn decode_hex_envelopes(
    hexes: &[String],
    idl: Option<&std::collections::HashMap<[u8; 4], String>>,
) -> Vec<Result<DecodedEnvelope, EventError>> {
    let mut out = Vec::with_capacity(hexes.len());
    for h in hexes {
        out.push(decode_hex_envelope(h, idl));
    }
    out
}

// crates/lez-events/src/batch.rs
use alloc::vec::Vec;
use borsh::BorshSerialize;

use crate::{encode_event_into, EventError, EventSchema};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BatchEnvelope {
    pub count: usize,
    pub total_bytes: usize,
    pub bytes: Vec<u8>,
}

#[derive(Debug, Clone)]
pub struct BatchEncoder {
    scratch: Vec<u8>,
    max_events: usize,
    max_bytes: usize,
}

impl Default for BatchEncoder {
    fn default() -> Self {
        Self {
            scratch: Vec::with_capacity(1024),
            max_events: crate::MAX_EVENTS_PER_TX,
            max_bytes: crate::MAX_TX_EVENT_BYTES,
        }
    }
}

impl BatchEncoder {
    pub fn with_limits(max_events: usize, max_bytes: usize) -> Self {
        Self {
            scratch: Vec::with_capacity(1024),
            max_events,
            max_bytes,
        }
    }

    pub fn clear(&mut self) {
        self.scratch.clear();
    }

    pub fn encode_one<E: EventSchema>(&mut self, event: &E) -> Result<&[u8], EventError> {
        encode_event_into(event, &mut self.scratch)?;
        Ok(&self.scratch)
    }

    pub fn push_encoded(&mut self, encoded: &[u8]) -> Result<(), EventError> {
        if encoded.len() > self.max_bytes {
            return Err(EventError::EventTooLarge {
                size: encoded.len(),
                limit: self.max_bytes,
            });
        }
        if encoded.len() + self.scratch.len() > self.max_bytes {
            return Err(EventError::TxBudgetExceeded {
                used: self.scratch.len(),
                added: encoded.len(),
                limit: self.max_bytes,
            });
        }

        self.scratch.extend_from_slice(encoded);
        Ok(())
    }

    pub fn finish(self, count: usize) -> Result<BatchEnvelope, EventError> {
        if count > self.max_events {
            return Err(EventError::TxCountExceeded {
                used: count,
                limit: self.max_events,
            });
        }

        Ok(BatchEnvelope {
            count,
            total_bytes: self.scratch.len(),
            bytes: self.scratch,
        })
    }

    pub fn encode_batch<E, I>(&mut self, events: I) -> Result<BatchEnvelope, EventError>
    where
        E: EventSchema,
        I: IntoIterator<Item = E>,
    {
        self.clear();
        let mut count = 0usize;

        for event in events {
            if count >= self.max_events {
                return Err(EventError::TxCountExceeded {
                    used: count,
                    limit: self.max_events,
                });
            }

            let encoded = crate::encode_event(&event)?;
            if self.scratch.len() + encoded.len() > self.max_bytes {
                return Err(EventError::TxBudgetExceeded {
                    used: self.scratch.len(),
                    added: encoded.len(),
                    limit: self.max_bytes,
                });
            }

            self.scratch.extend_from_slice(&encoded);
            count += 1;
        }

        self.finish(count)
    }
}

impl BatchEnvelope {
    pub fn as_slice(&self) -> &[u8] {
        &self.bytes
    }
}

    Failed,

    pub tx_hash: String,
    pub status: ReceiptStatus,
    pub error: Option<String>,
    pub state_root: Option<String>,

pub struct DecodedReceipt {
    pub tx_hash: String,
    pub status: ReceiptStatus,
    pub error: Option<String>,
    pub state_root: Option<String>,
    pub events: Vec<crate::DecodedEnvelope>,
}

impl ReceiptEnvelope {
    pub fn is_success(&self) -> bool {
        matches!(self.status, ReceiptStatus::Success)
    }

    pub fn event_count(&self) -> usize {
        self.events.len()
    }

    pub fn total_hex_chars(&self) -> usize {
        self.events.iter().map(|e| e.len()).sum()
    }

    pub fn validate(&self) -> Result<(), crate::EventError> {
        crate::validate_tx_hash(&self.tx_hash)?;
        for evt in &self.events {
            crate::validate_hex_string(evt)?;
        }
        Ok(())
    }
}

use alloc::string::String;
use sha2::{Digest, Sha256};

pub fn checksum_hex(bytes: &[u8]) -> String {
    let mut hasher = Sha256::new();
    hasher.update(bytes);
    hex::encode(hasher.finalize())
}

pub fn verify_checksum_hex(bytes: &[u8], checksum_hex: &str) -> bool {
    checksum_hex(bytes) == checksum_hex.trim().trim_start_matches("0x")
}

impl Backoff {
    pub fn new(cfg: BackoffConfig) -> Self {
        Self {
            cfg,
            attempt: 0,
            elapsed_ms: 0,
            seed: 0x9E37_79B9_7F4A_7C15,
        }
    }

    fn next_jitter(&mut self, base: u64) -> u64 {
        self.seed ^= self.seed << 13;
        self.seed ^= self.seed >> 7;
        self.seed ^= self.seed << 17;
        let range = (base / 4).max(1);
        let offset = self.seed % (range * 2 + 1);
        base.saturating_sub(range) + offset
    }

    pub fn next_delay(&mut self) -> Option<Duration> {
        if self.elapsed_ms >= self.cfg.max_elapsed_ms {
            return None;
        }

        let mut delay = self.cfg.initial_delay_ms;
        for _ in 0..self.attempt {
            delay = delay.saturating_mul(self.cfg.multiplier as u64);
            if delay > self.cfg.max_delay_ms {
                delay = self.cfg.max_delay_ms;
                break;
            }
        }

        if self.cfg.jitter {
            delay = self.next_jitter(delay).min(self.cfg.max_delay_ms);
        }

        self.attempt += 1;
        self.elapsed_ms = self.elapsed_ms.saturating_add(delay);
        Some(Duration::from_millis(delay))
    }

    pub fn reset(&mut self) {
        self.attempt = 0;
        self.elapsed_ms = 0;
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub backoff: BackoffConfig,
}

impl Default for RetryConfig {

        Self {
            attempts: 5,
            backoff: BackoffConfig::default(),
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct RetryError {
    pub attempts: usize,
    pub last_error: String,
}

impl core::fmt::Display for RetryError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        write!(
            f,
            "retry exhausted after {} attempts: {}",
            self.attempts, self.last_error
        )
    }
}

#[cfg(feature = "std")]
impl std::error::Error for RetryError {}

pub fn retry<T, F>(cfg: RetryConfig, mut op: F) -> Result<T, EventError>
where
    F: FnMut() -> Result<T, EventError>,
{
    let mut backoff = Backoff::new(cfg.backoff);
    let mut last_error = None;

    for attempt in 0..cfg.attempts {
        match op() {
            Ok(v) => return Ok(v),
            Err(err) => {
                last_error = Some(err.to_string());
                if attempt + 1 >= cfg.attempts {
                    break;
                }
                if let Some(_delay) = backoff.next_delay() {
                    // In CLI builds this is intentionally a hook point:
                    // plug in sleep here, or keep it no-op for testability.
                }
            }
        }
    }

    Err(EventError::RetryExhausted {
        attempts: cfg.attempts,
        last_error: last_error.unwrap_or_else(|| "unknown error".into()),
    })
}

pub fn atomic_write_bytes(path: &Path, bytes: &[u8]) -> Result<(), EventError> {
    let parent = path.parent().unwrap_or_else(|| Path::new("."));
    fs::create_dir_all(parent).map_err(|e| EventError::Io(e.to_string()))?;

    let mut tmp = path.to_path_buf();
    let file_name = path
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("tmp");
    tmp.set_file_name(format!(".{}.tmp", file_name));

    {
        let mut file = BufWriter::new(
            fs::File::create(&tmp).map_err(|e| EventError::Io(e.to_string()))?,
        );
        file.write_all(bytes)
            .map_err(|e| EventError::Io(e.to_string()))?;
        file.flush()
            .map_err(|e| EventError::Io(e.to_string()))?;
    }

    fs::rename(&tmp, path).map_err(|e| EventError::Io(e.to_string()))?;
    Ok(())
}

pub fn atomic_write_json<T: Serialize>(path: &Path, value: &T) -> Result<(), EventError> {
    let bytes = serde_json::to_vec_pretty(value).map_err(|e| EventError::Io(e.to_string()))?;
    atomic_write_bytes(path, &bytes)
}

pub fn read_json_file<T: DeserializeOwned>(path: &Path) -> Result<T, EventError> {
    let raw = fs::read_to_string(path).map_err(|e| EventError::Io(e.to_string()))?;
    serde_json::from_str(&raw).map_err(|e| EventError::Io(e.to_string()))
}

use alloc::string::String;

use crate::{DecodedEnvelope, ReceiptEnvelope, ReceiptStatus};

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct EventIndexEntry {
    pub tx_hash: String,
    pub status: ReceiptStatus,
    pub type_name: Option<String>,
    pub discriminant: String,
    pub payload_hex: String,
    pub payload_size: usize,
    pub raw_size: usize,
}

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct EventIndex {
    pub entries: Vec<EventIndexEntry>,
    pub by_tx: HashMap<String, Vec<usize>>,
    pub by_type: HashMap<String, Vec<usize>>,
}

#[derive(Debug, Clone, Default)]
pub struct EventIndexQuery {
    pub tx_hash: Option<String>,
    pub type_name: Option<String>,
    pub status: Option<ReceiptStatus>,
    pub limit: Option<usize>,
}

impl EventIndex {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn push_receipt(&mut self, receipt: &ReceiptEnvelope, decoded: &[DecodedEnvelope]) {
        let start = self.entries.len();
        for env in decoded {
            let idx = self.entries.len();
            self.entries.push(EventIndexEntry {
                tx_hash: receipt.tx_hash.clone(),
                status: receipt.status.clone(),
                type_name: env.type_name.clone(),
                discriminant: env.discriminant.clone(),
                payload_hex: env.payload_hex.clone(),
                payload_size: env.payload_size,
                raw_size: env.raw_size,
            });
            self.by_tx.entry(receipt.tx_hash.clone()).or_default().push(idx);
            if let Some(name) = env.type_name.clone() {
                self.by_type.entry(name).or_default().push(idx);
            }
        }
        let _ = start;
    }

    pub fn query(&self, q: &EventIndexQuery) -> Vec<&EventIndexEntry> {
        let mut out = Vec::new();
        for entry in &self.entries {
            if let Some(ref tx) = q.tx_hash {
                if &entry.tx_hash != tx {
                    continue;
                }
            }
            if let Some(ref ty) = q.type_name {
                if entry.type_name.as_deref() != Some(ty.as_str()) {
                    continue;
                }
            }
            if let Some(ref status) = q.status {
                if &entry.status != status {
                    continue;
                }
            }
            out.push(entry);
            if let Some(limit) = q.limit {
                if out.len() >= limit {
                    break;
                }
            }
        }
        out
    }
}

#[derive(Debug, Clone)]
pub struct EventJournal {
    tx_hash: String,
    events: Vec<String>,
    total_bytes: usize,
    count: usize,
    max_bytes: usize,
    max_count: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JournalError {
    BudgetExceeded { used: usize, added: usize, limit: usize },
    CountExceeded { used: usize, limit: usize },
}

impl core::fmt::Display for JournalError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::BudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: {used} + {added} > {limit}")
            }
            Self::CountExceeded { used, limit } => {
                write!(f, "event count exceeded: {used} >= {limit}")
            }
        }
    }
}

impl std::error::Error for JournalError {}

impl EventJournal {
    pub fn new(tx_hash: impl Into<String>) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes: crate::MAX_TX_EVENT_BYTES,
            max_count: crate::MAX_EVENTS_PER_TX,
        }
    }

    pub fn with_limits(tx_hash: impl Into<String>, max_bytes: usize, max_count: usize) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            total_bytes: 0,
            count: 0,
            max_bytes,
            max_count,
        }
    }

    pub fn push_hex(&mut self, hex_event: String, raw_len: usize) -> Result<(), JournalError> {
        if self.count >= self.max_count {
            return Err(JournalError::CountExceeded {
                used: self.count,
                limit: self.max_count,
            });
        }

        let next = self.total_bytes.saturating_add(raw_len);
        if next > self.max_bytes {
            return Err(JournalError::BudgetExceeded {
                used: self.total_bytes,
                added: raw_len,
                limit: self.max_bytes,
            });
        }

        self.events.push(hex_event);
        self.count += 1;
        self.total_bytes = next;
        Ok(())
    }

    pub fn finish_success(self, state_root: impl Into<String>) -> EventReceipt {
        let checksum = checksum_hex(self.events.join("").as_bytes());
        EventReceipt {
            tx_hash: self.tx_hash,
            status: ReceiptStatus::Success,
            error: None,
            state_root: Some(state_root.into()),
            events: self.events,
            checksum_hex: Some(checksum),
        }
    }

    pub fn finish_failure(self, error: impl Into<String>) -> EventReceipt {
        let checksum = checksum_hex(self.events.join("").as_bytes());
        EventReceipt {
            tx_hash: self.tx_hash,
            status: ReceiptStatus::Failed,
            error: Some(error.into()),
            state_root: None,
            events: self.events,
            checksum_hex: Some(checksum),
        }
    }
}

pub trait ReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError>;
    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError>;
    fn list(&self) -> Result<Vec<EventReceipt>, StoreError>;
}

#[derive(Debug, Clone)]
pub struct JsonFileReceiptStore {
    path: PathBuf,
}

impl JsonFileReceiptStore {
    pub fn new(path: impl Into<PathBuf>) -> Self {
        Self { path: path.into() }
    }

    fn load_all(&self) -> Result<Vec<EventReceipt>, StoreError> {
        if !self.path.exists() {
            return Ok(Vec::new());
        }

        let raw = fs::read_to_string(&self.path).map_err(StoreError::Io)?;
        if raw.trim().is_empty() {
            return Ok(Vec::new());
        }

        serde_json::from_str(&raw).map_err(StoreError::Serde)
    }

    fn save_all(&self, receipts: &[EventReceipt]) -> Result<(), StoreError> {
        if let Some(parent) = self.path.parent() {
            fs::create_dir_all(parent).map_err(StoreError::Io)?;
        }

        let tmp = self.path.with_extension("tmp");
        {
            let file = File::create(&tmp).map_err(StoreError::Io)?;
            let mut writer = BufWriter::new(file);
            serde_json::to_writer_pretty(&mut writer, receipts).map_err(StoreError::Serde)?;
            writer.flush().map_err(StoreError::Io)?;
        }

        fs::rename(tmp, &self.path).map_err(StoreError::Io)?;
        Ok(())
    }
}

impl ReceiptStore for JsonFileReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError> {
        let mut receipts = self.load_all()?;
        if let Some(existing) = receipts.iter_mut().find(|r| r.tx_hash == receipt.tx_hash) {
            *existing = receipt.clone();
        } else {
            receipts.push(receipt.clone());
        }
        self.save_all(&receipts)
    }

    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError> {
        Ok(self.load_all()?.into_iter().find(|r| r.tx_hash == tx_hash))
    }

    fn list(&self) -> Result<Vec<EventReceipt>, StoreError> {
        self.load_all()
    }
}

#[derive(Debug)]
pub enum StoreError {
    Io(std::io::Error),
    Serde(serde_json::Error),
}

impl core::fmt::Display for StoreError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Io(e) => write!(f, "i/o error: {e}"),
            Self::Serde(e) => write!(f, "serialization error: {e}"),
        }
    }
}

impl std::error::Error for StoreError {}

pub fn resolve_config(cli: &CommandLine) -> Result<CliConfig> {
    let mut cfg = if let Some(path) = cli.config.as_ref() {
        load_config_file(path)?
    } else if let Ok(path) = env::var("LEZ_EVENTS_CONFIG") {
        load_config_file(PathBuf::from(path))?
    } else {
        CliConfig::default()
    };

    if let Some(rpc) = cli.rpc_url.as_ref() {
        cfg.rpc_url = rpc.clone();
    }

    if cli.json {
        cfg.output = OutputFormat::Json;
    }
    if cli.jsonl {
        cfg.output = OutputFormat::JsonLines;
    }
    if cli.pretty {
        cfg.output = OutputFormat::Pretty;
    }

    if cli.no_color {
        cfg.color = false;
    }
    if cli.strict {
        cfg.strict = true;
    }
    if cli.follow {
        cfg.follow = true;
    }

    cfg.validate().context("invalid configuration")?;
    Ok(cfg)
}

pub fn render_error(err: &dyn std::error::Error) -> String {
    let mut out = err.to_string();
    let mut source = err.source();

    }
    out
}

fn index_command(args: IndexArgs, _cfg: CliConfig) -> Result<()> {
    let input = if let Some(path) = args.file {
        fs::read_to_string(&path).with_context(|| format!("failed to read {}", path.display()))?
    } else {
        prompts::read_stdin_all()?
    };

    let receipts: Vec<ReceiptEnvelope> = if input.trim().starts_with('[') {
        serde_json::from_str(&input)?
    } else {
        vec![serde_json::from_str(&input)?]
    };

    let mut decoded: Vec<DecodedReceipt> = Vec::with_capacity(receipts.len());
    let mut index = EventIndex::new();

    for receipt in receipts.into_iter().take(args.max_items) {
        receipt.validate().context("receipt failed validation")?;

        let parsed_events = receipt
            .events
            .iter()
            .map(|hex| decode_hex_envelope(hex, None))
            .collect::<Result<Vec<_>, _>>()
            .map_err(anyhow::Error::from)?;

        index.push_receipt(&receipt, &parsed_events);

        decoded.push(DecodedReceipt {
            tx_hash: receipt.tx_hash,
            status: receipt.status,
            error: receipt.error,
            state_root: receipt.state_root,
            events: parsed_events,
        });
    }

    atomic_write_json(&args.out, &decoded)?;
    println!("indexed {} receipts into {}", decoded.len(), args.out.display());
    Ok(())
}

fn query_command(args: QueryArgs, _cfg: CliConfig) -> Result<()> {
    let raw = fs::read_to_string(&args.file)
        .with_context(|| format!("failed to read {}", args.file.display()))?;
    let rows: Vec<DecodedReceipt> = serde_json::from_str(&raw)?;

    let filtered: Vec<_> = rows
        .into_iter()
        .filter(|r| args.tx_hash.as_ref().map_or(true, |wanted| &r.tx_hash == wanted))
        .filter(|r| args.status.as_ref().map_or(true, |wanted| format!("{:?}", r.status).to_lowercase() == wanted.to_lowercase()))
        .filter(|r| args.type_name.as_ref().map_or(true, |wanted| {
            r.events.iter().any(|e| e.type_name.as_deref() == Some(wanted))
        }))
        .collect();

    println!("{}", serde_json::to_string_pretty(&filtered)?);
    Ok(())
}

fn load_receipt_envelope(args: &DecodeArgs, cfg: &CliConfig) -> Result<ReceiptEnvelope> {
    if let Some(path) = &args.file {
        let raw = fs::read_to_string(path)
            .with_context(|| format!("failed to read {}", path.display()))?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)
            .with_context(|| format!("failed to parse {}", path.display()))?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    if args.stdin {
        let raw = prompts::read_stdin_all()?;
        let env: ReceiptEnvelope = serde_json::from_str(&raw)?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    if let Some(tx) = &args.tx {
        validate_tx_hash(tx)?;
        let rpc = args.rpc.as_deref().unwrap_or(&cfg.rpc_url);
        let client = Client::builder()
            .timeout(Duration::from_millis(cfg.timeout_ms))
            .build()
            .map_err(|e| EventError::Rpc(e.to_string()))?;
        let url = format!("{}/tx/{}/events", rpc.trim_end_matches('/'), tx);

        let resp = retry(
            RetryConfig {
                attempts: cfg.retries.max(1),
                backoff: lez_events::BackoffConfig::default(),
            },
            || {
                client
                    .get(&url)
                    .send()
                    .map_err(|e| EventError::Rpc(e.to_string()))
            },
        )?;

        if !resp.status().is_success() {
            return Err(EventError::Rpc(format!("rpc returned HTTP {}", resp.status())).into());
        }

        let env: ReceiptEnvelope = resp
            .json()
            .map_err(|e| EventError::Rpc(e.to_string()))?;
        env.validate().context("receipt failed validation")?;
        return Ok(env);
    }

    Err(EventError::MissingField("file | stdin | tx").into())
}

fn decode_receipt(envelope: &ReceiptEnvelope, args: &DecodeArgs) -> Result<DecodedReceipt> {
    let idl: HashMap<[u8; 4], String> = args
        .types
        .iter()
        .map(|t| (lez_events::event_discriminant(t), t.clone()))
        .collect();

    let mut decoded_events = Vec::with_capacity(envelope.events.len());
    for raw_hex in &envelope.events {
        match decode_hex_envelope(raw_hex, Some(&idl)) {
            Ok(e) => decoded_events.push(e),
            Err(err) => {
                if args.raw {
                    decoded_events.push(lez_events::DecodedEnvelope {
                        version: 0,
                        discriminant: "00000000".into(),
                        type_name: Some(format!("decode_error: {}", err)),
                        payload_hex: raw_hex.clone(),
                        payload_size: raw_hex.len() / 2,
                        raw_size: raw_hex.len(),
                    });
                } else {
                    return Err(err.into());
                }
            }
        }
    }

    Ok(DecodedReceipt {
        tx_hash: envelope.tx_hash.clone(),
        status: envelope.status.clone(),
        error: envelope.error.clone(),
        state_root: envelope.state_root.clone(),
        events: decoded_events,
    })
}

#[derive(Debug)]
pub enum ProgramError {
    Event(EventError),
    OversizeRequested { requested: usize, limit: usize },
    PanicSimulated(&'static str),
}

impl core::fmt::Display for ProgramError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Event(e) => write!(f, "{e}"),
            Self::OversizeRequested { requested, limit } => {
                write!(f, "requested event size {requested} exceeds limit {limit}")
            }
            Self::PanicSimulated(msg) => write!(f, "simulated failure: {msg}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for ProgramError {}

impl From<EventError> for ProgramError {
    fn from(value: EventError) -> Self {
        Self::Event(value)
    }
}

pub fn process_instruction(instruction: Instruction) -> Result<(), ProgramError> {
    match instruction {
        Instruction::Success { value } => {
            emit_event!(SuccessEvent { value })?;
            Ok(())
        }
        Instruction::FailAfterEvent { reason } => {
            emit_event!(FailureEvent { reason: reason.clone() })?;
            Err(ProgramError::PanicSimulated("intentional failure after event"))
        }
        Instruction::FailWithoutEvent => {
            Err(ProgramError::PanicSimulated("intentional failure before event emission"))
        }
        Instruction::SizeLimit => {
            let payload = vec![0u8; MAX_EVENT_SIZE + 1];
            let oversized = OversizedEvent { data: payload };
            match emit_event!(oversized) {
                Ok(()) => Ok(()),
                Err(EventError::EventTooLarge { size, limit }) => {
                    Err(ProgramError::OversizeRequested { requested: size, limit })
                }
                Err(err) => Err(ProgramError::Event(err)),
            }
        }
        Instruction::BatchSuccess { start, count } => {
            let mut current = start;
            let mut emitted = 0u32;
            while emitted < count {
                emit_event!(SuccessEvent { value: current })?;
                current = current.wrapping_add(1);
                emitted += 1;
            }
            Ok(())
        }
        Instruction::FriendlyMessage { message } => {
            emit_event!(FriendlyEvent { message })?;
            Ok(())
        }
        Instruction::Private { key, nonce, secret } => {
            emit_encrypted_event!(SecretEvent { secret }, &key, &nonce)
                .map_err(ProgramError::Event)
        }
    }
}
