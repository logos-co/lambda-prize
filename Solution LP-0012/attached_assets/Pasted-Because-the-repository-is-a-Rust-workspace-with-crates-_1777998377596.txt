function hashLike(input: string) {
  let h = 2166136261;
  for (let i = 0; i < input.length; i += 1) {
    h ^= input.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return Math.abs(h);
}

function pickWinner(seed: string, slot: number, stake: number, totalStake: number) {
  const score = hashLike(`${seed}:${slot}:${stake}:${totalStake}`) % 1_000_000;
  const threshold = Math.floor((stake / Math.max(totalStake, 1)) * 300_000);
  return { score, threshold, won: score <= threshold };
}

export default function SimulatorPage() {
  const [seed, setSeed] = useState("logos-seed");
  const [stake, setStake] = useState(12_500);
  const [totalStake, setTotalStake] = useState(250_000);
  const [slots, setSlots] = useState(8);

  const rows = useMemo(() => {
    return Array.from({ length: slots }, (_, i) => {
      const outcome = pickWinner(seed, i + 1, stake, totalStake);
      return {
        slot: i + 1,
        ...outcome
      };
    });
  }, [seed, stake, totalStake, slots]);

  return (
    <AppShell activePath="/simulator">
      <PageHero
        badge="Simulator"
        title="A tiny leadership lottery preview"
        description="This mock page helps visitors see how a slot-by-slot selection story could feel in the browser."
