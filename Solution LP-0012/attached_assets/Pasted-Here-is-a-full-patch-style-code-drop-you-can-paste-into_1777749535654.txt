extern crate alloc;

use alloc::string::{String, ToString};
use alloc::vec::Vec;
use borsh::BorshSerialize;

/// Maximum serialised payload of a single event.
pub const MAX_EVENT_SIZE: usize = 64 * 1024;

/// Maximum total event bytes per transaction.
pub const MAX_TX_EVENT_BYTES: usize = 1024 * 1024;

/// Maximum number of events per transaction.
pub const MAX_EVENTS_PER_TX: usize = 256;

/// Wire-format version byte.
pub const EVENT_VERSION: u8 = 0x00;

/// Event type contract.
///
/// Every event must provide a stable, fully-qualified type name.
/// That string is hashed into a 4-byte discriminant.
pub trait EventSchema: BorshSerialize {
    const NAME: &'static str;
    const DISCRIMINANT: [u8; 4] = fnv1a_discriminant(Self::NAME);

    fn encode(&self) -> Result<Vec<u8>, EventError>
    where
        Self: Sized,
    {
        encode_event(self, Self::NAME)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventError {
    SerializationFailed,
    EventTooLarge { size: usize, limit: usize },
    TxBudgetExceeded,
    TxCountExceeded,
    InvalidEncryptionKey,
    InvalidEncryptionNonce,
    EncryptionFailed,
    SyscallError(i32),
}

impl core::fmt::Display for EventError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::SerializationFailed => f.write_str("event serialization failed"),
            Self::EventTooLarge { size, limit } => {
                write!(f, "event too large: {size} B > {limit} B")
            }
            Self::TxBudgetExceeded => {
                write!(f, "transaction event byte budget exceeded (>{MAX_TX_EVENT_BYTES})")
            }
            Self::TxCountExceeded => {
                write!(f, "transaction event count exceeded (>{MAX_EVENTS_PER_TX})")
            }
            Self::InvalidEncryptionKey => f.write_str("invalid encryption key"),
            Self::InvalidEncryptionNonce => f.write_str("invalid encryption nonce"),
            Self::EncryptionFailed => f.write_str("event encryption failed"),
            Self::SyscallError(code) => write!(f, "sys_emit_event returned error code {code}"),
        }
    }
}

/// FNV-1a discriminant over the UTF-8 bytes of a type name.
pub const fn fnv1a_discriminant(name: &str) -> [u8; 4] {
    let bytes = name.as_bytes();
    let mut h: u32 = 0x811c_9dc5;
    let mut i = 0;
    while i < bytes.len() {
        h ^= bytes[i] as u32;
        h = h.wrapping_mul(0x0100_0193);
        i += 1;
    }
    h.to_le_bytes()
}

pub fn encode_event<E: BorshSerialize>(event: &E, type_name: &str) -> Result<Vec<u8>, EventError> {
    let payload = borsh::to_vec(event).map_err(|_| EventError::SerializationFailed)?;
    if payload.len() > MAX_EVENT_SIZE {
        return Err(EventError::EventTooLarge {
            size: payload.len(),
            limit: MAX_EVENT_SIZE,
        });
    }

    let disc = fnv1a_discriminant(type_name);
    let mut out = Vec::with_capacity(1 + 4 + payload.len());
    out.push(EVENT_VERSION);
    out.extend_from_slice(&disc);
    out.extend_from_slice(&payload);
    Ok(out)
}

#[macro_export]
macro_rules! emit_event {
    ($event:expr) => {{
        use $crate::EventSchema;
        let bytes = <$crate::__private::TypeOf<$event> as EventSchema>::encode(&$event)?;
        let rc = unsafe { $crate::__sys_emit_event(bytes.as_ptr(), bytes.len()) };
        match rc {
            0 => Ok(()),
            -1 => Err($crate::EventError::TxBudgetExceeded),
            -2 => Err($crate::EventError::TxCountExceeded),
            c => Err($crate::EventError::SyscallError(c)),
        }
    }};
    ($event:expr, $name:expr) => {{
        let bytes = $crate::encode_event(&$event, $name)?;
        let rc = unsafe { $crate::__sys_emit_event(bytes.as_ptr(), bytes.len()) };
        match rc {
            0 => Ok(()),
            -1 => Err($crate::EventError::TxBudgetExceeded),
            -2 => Err($crate::EventError::TxCountExceeded),
            c => Err($crate::EventError::SyscallError(c)),
        }
    }};
}

macro_rules! emit_encrypted_event {
    ($event:expr, $key_bytes:expr, $nonce_bytes:expr) => {{
        use aes_gcm::{Aes256Gcm, Key, Nonce, aead::{Aead, KeyInit}};
        use $crate::{EventError, EventSchema};

        let plaintext = <$crate::__private::TypeOf<$event> as EventSchema>::encode(&$event)?;
        let key = Key::<Aes256Gcm>::from_slice($key_bytes);
        let nonce = Nonce::from_slice($nonce_bytes);
        let cipher = Aes256Gcm::new(key);
        let ct = cipher.encrypt(nonce, plaintext.as_ref())
            .map_err(|_| EventError::EncryptionFailed)?;
        let rc = unsafe { $crate::__sys_emit_event(ct.as_ptr(), ct.len()) };
        match rc {
            0 => Ok(()),
            -1 => Err(EventError::TxBudgetExceeded),
            -2 => Err(EventError::TxCountExceeded),
            c => Err(EventError::SyscallError(c)),
        }
    }};

            event,
            type_name,
            allow_oversize: false,
        }
    }

    pub fn allow_oversize(mut self, yes: bool) -> Self {
        self.allow_oversize = yes;
        self
    }

        if !self.allow_oversize && bytes.len() > (1 + 4 + MAX_EVENT_SIZE) {
            return Err(EventError::EventTooLarge {
                size: bytes.len(),
                limit: 1 + 4 + MAX_EVENT_SIZE,
            });
        }
        Ok(bytes)
    }

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct DecodedEvent {
    pub version: u8,
    pub discriminant: [u8; 4],
    pub payload: Vec<u8>,
    pub payload_hex: String,
    pub type_name: Option<String>,
    pub program_id: Option<String>,
}

pub fn decode_raw(
    bytes: &[u8],
    idl: Option<&HashMap<[u8; 4], String>>,
) -> Option<DecodedEvent> {
    if bytes.len() < 5 {
        return None;
    }

    let version = bytes[0];
    let discriminant: [u8; 4] = bytes[1..5].try_into().ok()?;
    let payload = bytes[5..].to_vec();
    let payload_hex = hex::encode(&payload);
    let type_name = idl.and_then(|m| m.get(&discriminant)).cloned();

    Some(DecodedEvent {
        version,
        discriminant,
        payload,
        payload_hex,
        type_name,
        program_id: None,
    })
}

pub fn build_idl(entries: &[(&str, [u8; 4])]) -> HashMap<[u8; 4], String> {
    entries
        .iter()
        .map(|(name, disc)| (*disc, (*name).to_string()))
        .collect()
}

pub fn register_type(idl: &mut HashMap<[u8; 4], String>, type_name: &str) {
    idl.insert(crate::fnv1a_discriminant(type_name), type_name.to_string());
}

pub fn register_schema<E>(idl: &mut HashMap<[u8; 4], String>)
where
    E: crate::EventSchema,
{
    idl.insert(E::DISCRIMINANT, E::NAME.to_string());

    pub status: String,
    pub error: Option<String>,
    pub events: Vec<String>,

pub fn append_jsonl(path: &Path, entry: &LogEntry) -> std::io::Result<()> {
    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }
    let mut file = OpenOptions::new().create(true).append(true).open(path)?;
    serde_json::to_writer(&mut file, entry)?;
    file.write_all(b"\n")?;
    file.flush()
}

pub fn read_jsonl(path: &Path) -> std::io::Result<Vec<LogEntry>> {
    let file = File::open(path)?;
    let reader = BufReader::new(file);
    let mut out = Vec::new();

    for line in reader.lines() {
        let line = line?;
        if line.trim().is_empty() {
            continue;
        }
        let parsed: LogEntry = serde_json::from_str(&line)
            .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e))?;
        out.push(parsed);
    }

    Ok(out)
}

// crates/lez-events/src/merkle.rs
use alloc::vec::Vec;
use sha2::{Digest, Sha256};

pub fn sha256(bytes: &[u8]) -> [u8; 32] {
    let mut hasher = Sha256::new();
    hasher.update(bytes);
    hasher.finalize().into()
}

pub fn merkle_root(leaves: &[Vec<u8>]) -> [u8; 32] {
    if leaves.is_empty() {
        return [0u8; 32];
    }

    let mut level: Vec<[u8; 32]> = leaves.iter().map(|leaf| sha256(leaf)).collect();

    while level.len() > 1 {
        if level.len() % 2 == 1 {
            let last = *level.last().unwrap();
            level.push(last);
        }

        let mut next = Vec::with_capacity(level.len() / 2);
        let mut i = 0;
        while i < level.len() {
            let mut data = Vec::with_capacity(64);
            data.extend_from_slice(&level[i]);
            data.extend_from_slice(&level[i + 1]);
            next.push(sha256(&data));
            i += 2;
        }
        level = next;
    }

    level[0]
}

#[derive(Debug, Clone)]
pub struct EventJournal {
    tx_hash: String,
    events: Vec<String>,
    byte_budget: usize,
    event_count: usize,
    max_bytes: usize,
    max_count: usize,
}

impl EventJournal {
    pub fn new(tx_hash: impl Into<String>) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            byte_budget: 0,
            event_count: 0,
            max_bytes: crate::MAX_TX_EVENT_BYTES,
            max_count: crate::MAX_EVENTS_PER_TX,
        }
    }

    pub fn with_limits(tx_hash: impl Into<String>, max_bytes: usize, max_count: usize) -> Self {
        Self {
            tx_hash: tx_hash.into(),
            events: Vec::new(),
            byte_budget: 0,
            event_count: 0,
            max_bytes,
            max_count,
        }
    }

    pub fn push_hex(&mut self, hex_event: String, raw_len: usize) -> Result<(), JournalError> {
        if self.event_count >= self.max_count {
            return Err(JournalError::CountExceeded {
                limit: self.max_count,
            });
        }

        let next = self.byte_budget.saturating_add(raw_len);
        if next > self.max_bytes {
            return Err(JournalError::BudgetExceeded {
                used: self.byte_budget,
                added: raw_len,
                limit: self.max_bytes,
            });
        }

        self.events.push(hex_event);
        self.event_count += 1;
        self.byte_budget = next;
        Ok(())
    }

    pub fn finish_success(self, state_root: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: TransactionStatus::Success,
            error: None,
            state_root: Some(state_root.into()),
            events: self.events,
        }
    }

    pub fn finish_failure(self, error: impl Into<String>) -> EventReceipt {
        EventReceipt {
            tx_hash: self.tx_hash,
            status: TransactionStatus::Failed,
            error: Some(error.into()),
            state_root: None,
            events: self.events,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JournalError {
    BudgetExceeded { used: usize, added: usize, limit: usize },
    CountExceeded { limit: usize },
}

impl core::fmt::Display for JournalError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::BudgetExceeded { used, added, limit } => {
                write!(f, "event budget exceeded: used={used}, added={added}, limit={limit}")
            }
            Self::CountExceeded { limit } => write!(f, "event count exceeded: limit={limit}"),
        }
    }
}

impl std::error::Error for JournalError {}

pub trait ReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError>;
    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError>;
    fn list(&self) -> Result<Vec<EventReceipt>, StoreError>;
}

#[derive(Debug, Clone)]
pub struct JsonFileReceiptStore {
    path: PathBuf,
}

impl JsonFileReceiptStore {
    pub fn new(path: impl Into<PathBuf>) -> Self {
        Self { path: path.into() }
    }

    fn load_all(&self) -> Result<Vec<EventReceipt>, StoreError> {
        if !self.path.exists() {
            return Ok(Vec::new());
        }
        let raw = fs::read_to_string(&self.path).map_err(StoreError::Io)?;
        if raw.trim().is_empty() {
            return Ok(Vec::new());
        }
        serde_json::from_str(&raw).map_err(StoreError::Serde)
    }

    fn save_all(&self, receipts: &[EventReceipt]) -> Result<(), StoreError> {
        if let Some(parent) = self.path.parent() {
            fs::create_dir_all(parent).map_err(StoreError::Io)?;
        }
        let raw = serde_json::to_string_pretty(receipts).map_err(StoreError::Serde)?;
        fs::write(&self.path, raw).map_err(StoreError::Io)
    }
}

impl ReceiptStore for JsonFileReceiptStore {
    fn put(&mut self, receipt: &EventReceipt) -> Result<(), StoreError> {
        let mut receipts = self.load_all()?;
        if let Some(existing) = receipts.iter_mut().find(|r| r.tx_hash == receipt.tx_hash) {
            *existing = receipt.clone();
        } else {
            receipts.push(receipt.clone());
        }
        self.save_all(&receipts)
    }

    fn get(&self, tx_hash: &str) -> Result<Option<EventReceipt>, StoreError> {
        Ok(self.load_all()?.into_iter().find(|r| r.tx_hash == tx_hash))
    }

    fn list(&self) -> Result<Vec<EventReceipt>, StoreError> {
        self.load_all()
    }
}

#[derive(Debug)]
pub enum StoreError {
    Io(std::io::Error),
    Serde(serde_json::Error),
}

impl core::fmt::Display for StoreError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Io(e) => write!(f, "io error: {e}"),
            Self::Serde(e) => write!(f, "serialization error: {e}"),
        }
    }
}

impl std::error::Error for StoreError {}

    let cli = Cli::parse();

    match cli.command {
        Commands::Decode {
            file,
            tx,
            rpc,
            pretty,
            types,
            key,
            nonce,
        } => {
            let receipt = load_receipt(file, tx, rpc)?;
            let map = build_idl_map(&types);
            let decoded = decode_receipt(&receipt, &map, key.as_deref(), nonce.as_deref())?;
            let rendered = if pretty {
                serde_json::to_string_pretty(&decoded)?
            } else {
                serde_json::to_string(&decoded)?
            };
            println!("{rendered}");
        }

        Commands::Index { rpc, out, interval } => {
            index_loop(&rpc, &out, interval)?;
        }

        Commands::Query { file, tx, status } => {
            let receipts = read_receipts(&file)?;
            let filtered: Vec<_> = receipts
                .into_iter()
                .filter(|r| tx.as_ref().map_or(true, |wanted| &r.tx_hash == wanted))
                .filter(|r| status.as_ref().map_or(true, |wanted| &r.status == wanted))
                .collect();
            println!("{}", serde_json::to_string_pretty(&filtered)?);
        }
    }

    Ok(())
}

fn load_receipt(
    file: Option<PathBuf>,
    tx: Option<String>,
    rpc: Option<String>,
) -> Result<RawReceiptFile, Box<dyn std::error::Error>> {
    if let Some(path) = file {
        return Ok(serde_json::from_str(&fs::read_to_string(path)?)?);
    }

    let tx = tx.ok_or("missing --tx")?;
    let rpc = rpc.ok_or("missing --rpc")?;
    let _ = rpc;

    Err(format!(
        "RPC fetch is not wired in this minimal CLI sample; fetch tx {tx} into a local receipt JSON file first"
    )
    .into())
}

fn read_receipts(path: &PathBuf) -> Result<Vec<RawReceiptFile>, Box<dyn std::error::Error>> {
    let raw = fs::read_to_string(path)?;
    let receipts: Vec<RawReceiptFile> = serde_json::from_str(&raw)?;
    Ok(receipts)
}

fn build_idl_map(types: &[String]) -> HashMap<[u8; 4], String> {
    let mut idl = HashMap::new();
    for t in types {
        register_name(&mut idl, t);
    }
    idl
}

fn register_name(idl: &mut HashMap<[u8; 4], String>, type_name: &str) {
    idl.insert(lez_events::fnv1a_discriminant(type_name), type_name.to_string());
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct DecodedReceipt {
    tx_hash: String,
    status: String,
    error: Option<String>,
    state_root: Option<String>,
    events: Vec<serde_json::Value>,
}

fn decode_receipt(
    receipt: &RawReceiptFile,
    idl: &HashMap<[u8; 4], String>,
    key: Option<&str>,
    nonce: Option<&str>,
) -> Result<DecodedReceipt, Box<dyn std::error::Error>> {
    let mut out_events = Vec::new();

    for hex_event in &receipt.events {
        let raw = hex::decode(hex_event)?;
        if let Some(decoded) = decode_raw(&raw, Some(idl)) {
            let event_json = serde_json::json!({
                "version": decoded.version,
                "discriminant": hex::encode(decoded.discriminant),
                "type_name": decoded.type_name,
                "payload_hex": decoded.payload_hex,
                "program_id": decoded.program_id,
            });
            out_events.push(event_json);
        } else {
            out_events.push(serde_json::json!({
                "error": "malformed event envelope",
                "raw_hex": hex_event,
            }));
        }
    }

    Ok(DecodedReceipt {
        tx_hash: receipt.tx_hash.clone(),
        status: receipt.status.clone(),
        error: receipt.error.clone(),
        state_root: receipt.state_root.clone(),
        events: out_events,
    })
}

fn index_loop(
    rpc: &str,
    out: &PathBuf,
    interval: u64,
) -> Result<(), Box<dyn std::error::Error>> {
    let _ = rpc;
    let _ = interval;

    if !out.exists() {
        fs::write(out, "[]")?;
    }

    println!(
        "Indexer stub started. In a real deployment this would poll {rpc} and append to {out:?}."
    );
    Ok(())
}

#[derive(Debug)]
pub enum ProgramError {
    Event(EventError),
    MissingInput,
    OversizeRequested { requested: usize, limit: usize },
    PanicSimulated(&'static str),
}

impl core::fmt::Display for ProgramError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::Event(e) => write!(f, "{e}"),
            Self::MissingInput => f.write_str("missing program input"),
            Self::OversizeRequested { requested, limit } => {
                write!(f, "requested event size {requested} exceeds limit {limit}")
            }
            Self::PanicSimulated(msg) => write!(f, "simulated failure: {msg}"),
        }
    }
}

#[cfg(feature = "std")]
impl std::error::Error for ProgramError {}

impl From<EventError> for ProgramError {
    fn from(value: EventError) -> Self {
        Self::Event(value)
    }
}

pub fn process_instruction(instruction: Instruction) -> Result<(), ProgramError> {
    match instruction {
        Instruction::Success { value } => {
            emit_event!(SuccessEvent { value })?;
            Ok(())
        }

        Instruction::FailAfterEvent { reason } => {
            emit_event!(FailureEvent { reason: reason.clone() })?;
            Err(ProgramError::PanicSimulated("intentional failure after event"))
        }

        Instruction::FailWithoutEvent => Err(ProgramError::PanicSimulated(
            "intentional failure before event emission",
        )),

        Instruction::SizeLimit => {
            let payload = vec![0u8; MAX_EVENT_SIZE + 1];
            let oversized = OversizedEvent { data: payload };
            match emit_event!(oversized) {
                Ok(()) => Ok(()),
                Err(EventError::EventTooLarge { size, limit }) => {
                    Err(ProgramError::OversizeRequested { requested: size, limit })
                }
                Err(e) => Err(ProgramError::Event(e)),
            }
        }

        Instruction::Private { key, nonce, secret } => {
            emit_encrypted_event!(SecretEvent { secret }, &key, &nonce)
                .map_err(ProgramError::Event)
        }
    }
}

# LEZ Event Format Specification v1.1

## Overview

Each event emitted by a LEZ program is stored in a transaction receipt as a binary envelope with deterministic encoding and explicit attribution.

## Envelope

```text
[program_id: 32 bytes]
[version: 1 byte]
[discriminant: 4 bytes]
[payload: variable]

Encoding Rules
program_id is added by the runtime.
version is 0x00 for this revision.
discriminant is fnv1a_32(type_name)[0..4].
payload is Borsh-serialized.
Borsh uses little-endian integers, length-prefixed strings, and declaration-order struct fields.
Limits
Single event payload: 64 KiB.
Total event bytes per transaction: 1 MiB.
Events per transaction: 256.
Runtime Behavior
The runtime must reject an event immediately when any limit is exceeded. It must not truncate or silently drop bytes.
0 success
-1 transaction byte budget exceeded
-2 transaction event count exceeded
-3 event payload exceeds 64 KiB
Failure Semantics
A failed transaction keeps all events emitted before the failure point. The receipt therefore always contains the events array, even when status == failed.
Receipt Schema
{
  "tx_hash": "0x...",
  "status": "success | failed",
  "error": "optional error message",
  "state_root": "0x... (success only)",
  "events": ["hex-encoded envelope bytes"]
}

Decoder Guidance
Clients should:
Strip the first 32 bytes for program_id.
Read the version byte.
Read the 4-byte discriminant.
Decode the payload with the event schema known to the client.
Unknown version bytes should fail closed.
Encrypted Events
When encryption is enabled, the payload portion of the envelope may be an authenticated ciphertext. Clients without the key should treat the bytes as opaque.
